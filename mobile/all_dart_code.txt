abstract class AuthRemoteDataSource {
  Future<AuthResponseModel> login(LoginRequestModel loginRequest);
  Future<String> register(RegisterRequestModel registerRequest);
  Future<String> verifySms(VerifySmsRequestModel verifySmsRequest);
  Future<String> resendOtp(ResendOtpRequestModel resendOtpRequest);
}

class AuthRemoteDataSourceImpl implements AuthRemoteDataSource {
  final ApiClient apiClient;

  AuthRemoteDataSourceImpl({required this.apiClient});

  @override
  Future<AuthResponseModel> login(LoginRequestModel loginRequest) async {
    final response = await apiClient.post(
      ApiEndpoints.login,
      data: loginRequest.toJson(),
    );
    return AuthResponseModel.fromJson(response.data);
  }

  @override
  Future<String> register(RegisterRequestModel registerRequest) async {
    final response = await apiClient.post(
      ApiEndpoints.register,
      data: registerRequest.toJson(),
    );
    return response.data as String;
  }

  @override
  Future<String> verifySms(VerifySmsRequestModel verifySmsRequest) async {
    final response = await apiClient.post(
      ApiEndpoints.verifySms,
      data: verifySmsRequest.toJson(),
    );
    return response.data as String;
  }

  @override
  Future<String> resendOtp(ResendOtpRequestModel resendOtpRequest) async {
    final response = await apiClient.post(
      ApiEndpoints.resendOtp,
      data: resendOtpRequest.toJson(),
    );
    return response.data as String;
  }
}

import 'package:sahbifit_mobile/features/auth/data/datasources/auth_remote_datasource.dart';
import 'package:sahbifit_mobile/features/auth/data/models/login_request_model.dart';
import 'package:sahbifit_mobile/features/auth/data/models/register_request_model.dart';
import 'package:sahbifit_mobile/features/auth/data/models/resend_otp_request_model.dart';

import 'package:sahbifit_mobile/features/auth/domain/repositories/auth_repository.dart';

import '../models/auth_response_model.dart'; 

class AuthRepositoryImpl implements AuthRepository {
  final AuthRemoteDataSource remoteDataSource;

  AuthRepositoryImpl({required this.remoteDataSource});

  // Helper function to ensure phone number is in backend format
  String _stripCountryCode(String phoneNumber) {
    if (phoneNumber.startsWith('+216')) {
      return phoneNumber.substring(4); // Removes '+216'
    }
    // Add other prefixes if necessary, e.g., '00216'
    if (phoneNumber.startsWith('00216')) {
      return phoneNumber.substring(5); // Removes '00216'
    }
    // Assume it's already in the correct format if it doesn't match
    return phoneNumber;
  }

  @override
  Future<AuthResponseModel> login(String phoneNumber, String password) async {
    final formattedPhoneNumber = _stripCountryCode(phoneNumber);
    final request = LoginRequestModel(
      phoneNumber: formattedPhoneNumber,
      password: password,
    );
    return await remoteDataSource.login(request);
  }

  @override
  Future<String> register({
    required String phoneNumber,
    required String password,
    required String recaptchaToken,
    required String firstName,
    required String dateOfBirth,
    required String primaryGymId,
    String? profilePhotoUrl,
    required List<int> goalTagIds,
    required List<int> scheduleTagIds,
  }) async {
    final formattedPhoneNumber = _stripCountryCode(phoneNumber);
    final request = RegisterRequestModel(
      phoneNumber: formattedPhoneNumber,
      password: password,
      recaptchaToken: recaptchaToken,
      firstName: firstName,
      dateOfBirth: dateOfBirth,
      primaryGymId: primaryGymId,
      profilePhotoUrl: profilePhotoUrl,
      goalTagIds: goalTagIds,
      scheduleTagIds: scheduleTagIds,
    );
    return await remoteDataSource.register(request);
  }

  @override
  Future<String> verifySms(String phoneNumber, String otp) async {

import 'package:equatable/equatable.dart';

part 'auth_state.dart';

class AuthBloc extends Bloc<AuthEvent, AuthState> {
  final LoginUser loginUser;
  final RegisterUser registerUser;
  final VerifySms verifySms;
  final ResendOtp resendOtp;

  AuthBloc({
    required this.loginUser,
    required this.registerUser,
    required this.verifySms,
    required this.resendOtp,
  }) : super(AuthInitial()) {
    on<LoginRequested>(_onLoginRequested);
    on<RegisterRequested>(_onRegisterRequested);
    on<VerifySmsRequested>(_onVerifySmsRequested);
    on<ResendOtpRequested>(_onResendOtpRequested);
    on<LogoutRequested>(_onLogoutRequested);
  }

  Future<void> _onLoginRequested(
    LoginRequested event,
    Emitter<AuthState> emit,
  ) async {
    emit(AuthLoading());
    try {
      final token = await loginUser(event.phoneNumber, event.password);
      emit(AuthAuthenticated(token: token));
    } catch (e) {
      emit(AuthError(message: e.toString()));
    }
  }

  Future<void> _onRegisterRequested(
    RegisterRequested event,
    Emitter<AuthState> emit,
  ) async {
    emit(AuthLoading());
    try {
      final message = await registerUser.call(
        phoneNumber: event.phoneNumber,
        password: event.password,
        recaptchaToken: event.recaptchaToken,
        firstName: event.firstName,
        dateOfBirth: event.dateOfBirth,
        primaryGymId: event.primaryGymId,
        profilePhotoUrl: event.profilePhotoUrl,
        goalTagIds: event.goalTagIds,
        scheduleTagIds: event.scheduleTagIds,
      );
      emit(RegistrationSuccess(message: message));
      emit(AuthUnverified(phoneNumber: event.phoneNumber));
    } catch (e) {
      emit(AuthError(message: e.toString()));
    }
  }

  Future<void> _onVerifySmsRequested(
    VerifySmsRequested event,
    Emitter<AuthState> emit,
  ) async {
    emit(AuthLoading());
    try {
      final message = await verifySms(event.phoneNumber, event.otp);
      emit(VerificationSuccess(message: message));
      emit(AuthUnauthenticated());
    } catch (e) {
      emit(AuthError(message: e.toString()));
    }
  }

  Future<void> _onResendOtpRequested(
    ResendOtpRequested event,
    Emitter<AuthState> emit,
  ) async {
    emit(AuthLoading());
    try {
      final message = await resendOtp(event.phoneNumber, event.recaptchaToken);
      emit(OtpResent(message: message));
    } catch (e) {
      emit(AuthError(message: e.toString()));
    }
  }

  void _onLogoutRequested(
    LogoutRequested event,
    Emitter<AuthState> emit,
  ) {
    emit(AuthUnauthenticated());
  }
}

import 'package:flutter/material.dart';
import '../../../../../core/utils/icon_utils.dart';

class RatingStars extends StatelessWidget {
  final int rating;
  final Function(int) onRatingChanged;
  final double size;
  final bool interactive;

  const RatingStars({
    super.key,
    required this.rating,
    required this.onRatingChanged,
    this.size = 24,
    this.interactive = true,
  });

  @override
  Widget build(BuildContext context) {
    return Row(
      mainAxisSize: MainAxisSize.min,
      children: List.generate(5, (index) {
        final starRating = index + 1;
        return GestureDetector(
          onTap: interactive ? () => onRatingChanged(starRating) : null,
          child: Icon(
            starRating <= rating
                ? IconUtils.starIcon
                : IconUtils.starOutlineIcon,
            color: Colors.amber,
            size: size,
          ),
        );
      }),
    );
  }
}

class StaticRatingStars extends StatelessWidget {
  final double rating;
  final double size;

  const StaticRatingStars({
    super.key,
    required this.rating,
    this.size = 16,
  });

  @override
  Widget build(BuildContext context) {
    return Row(
      mainAxisSize: MainAxisSize.min,
      children: List.generate(5, (index) {
        if (index < rating.floor()) {
          return Icon(IconUtils.starIcon, color: Colors.amber, size: size);
        } else if (index == rating.floor() && rating % 1 != 0) {
          return Icon(IconUtils.starHalfIcon, color: Colors.amber, size: size);
        } else {
          return Icon(IconUtils.starOutlineIcon, color: Colors.amber, size: size);
        }
      }),
    );
  }
}

                        color: _getStatusColor(participant.status),
                      ),
                ),
              ],
            ),
          ),
          if (showAttendanceActions && onAttendanceChanged != null)
            _buildAttendanceToggle(participant),
        ],
      ),
    );
  }

  Widget _buildAttendanceToggle(SessionParticipant participant) {
    final isPresent = participant.status == ParticipantStatus.attended;
    
    return Row(
      mainAxisSize: MainAxisSize.min,
      children: [
        Text(
          isPresent ? 'Present' : 'Absent',
          style: TextStyle(
            color: isPresent ? Colors.green : Colors.red,
            fontWeight: FontWeight.w500,
          ),
        ),
        const SizedBox(width: 8),
        Switch(
          value: isPresent,
          onChanged: (value) {
            onAttendanceChanged!(participant.id, value);
          },
          activeColor: Colors.green,
        ),
      ],
    );
  }

  Color _getStatusColor(ParticipantStatus status) {
    switch (status) {
      case ParticipantStatus.attended:
        return Colors.green;
      case ParticipantStatus.registered:
        return Colors.blue;
      case ParticipantStatus.noShow:
        return Colors.red;
      case ParticipantStatus.pending:

  }

            if (state is TagsLoading)
              const Center(child: CircularProgressIndicator())
            else if (state is TagsLoaded)
              _buildTagGrid(state.tags)
            else if (state is TagsError)
              Text('Error: ${state.message}')
            else
              const SizedBox(),
          ],
        );
      },
    );

  Widget _buildTagGrid(List<Tag> tags) {
    return Wrap(
      spacing: 8,
      runSpacing: 8,
      children: tags.map((tag) {
        final isSelected = widget.selectedTags.contains(tag.id);
        return FilterChip(
          label: Text(tag.name),
          selected: isSelected,
          onSelected: (selected) {
            setState(() {
              if (selected) {
                widget.selectedTags.add(tag.id);
              } else {
                widget.selectedTags.remove(tag.id);
              }
              widget.onTagsChanged(widget.selectedTags);
            });
          },
          selectedColor: Colors.red.withOpacity(0.2),
          checkmarkColor: Colors.red,
          labelStyle: TextStyle(
            color: isSelected ? Colors.red : Colors.black87,
          ),
        );
      }).toList(),
    );
  }
}

import 'package:flutter/material.dart';
import 'package:camera/camera.dart';
import '../../../../../shared_widgets/primary_button.dart';

class VideoRecorderWidget extends StatefulWidget {
  final Function(String) onVideoRecorded;

  const VideoRecorderWidget({super.key, required this.onVideoRecorded});

  @override
  State<VideoRecorderWidget> createState() => _VideoRecorderWidgetState();
}

class _VideoRecorderWidgetState extends State<VideoRecorderWidget> {
  CameraController? _controller;
  bool _isRecording = false;
  bool _isCameraReady = false;

  @override
  void initState() {
    super.initState();
    _initializeCamera();
  }

  Future<void> _initializeCamera() async {
    final cameras = await availableCameras();
    final frontCamera = cameras.firstWhere(
      (camera) => camera.lensDirection == CameraLensDirection.front,
      orElse: () => cameras.first,
    );

    _controller = CameraController(
      frontCamera,
      ResolutionPreset.medium,
    );

    await _controller!.initialize();
    setState(() {
      _isCameraReady = true;
    });
  }

  @override
  Widget build(BuildContext context) {
    return Column(
      children: [
        Container(
          height: 300,
          decoration: BoxDecoration(
            borderRadius: BorderRadius.circular(16),
            color: Colors.black,
          ),
          child: _isCameraReady && _controller != null
              ? ClipRRect(
                  borderRadius: BorderRadius.circular(16),
                  child: CameraPreview(_controller!),
                )
              : const Center(child: CircularProgressIndicator()),
        ),
        const SizedBox(height: 24),
        Text(
          'Verification Video',
          style: Theme.of(context).textTheme.titleLarge,
        ),
        const SizedBox(height: 8),
        Text(
          'Please record a 5-second video saying: '
          '"Hi, I\'m [Your Name]. Today is [Date]." '
          'Then turn your head left and right.',
          textAlign: TextAlign.center,
          style: Theme.of(context).textTheme.bodyMedium,
        ),
        const SizedBox(height: 24),
        if (!_isRecording)
          PrimaryButton(
            text: 'Start Recording',
            onPressed: _startRecording,
            icon: Icons.videocam,
          )
        else
          Row(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              Icon(Icons.radio_button_checked, color: Colors.red),
              const SizedBox(width: 8),
              Text(
                'Recording...',
                style: Theme.of(context).textTheme.bodyLarge?.copyWith(
                      color: Colors.red,
                      fontWeight: FontWeight.w600,
                    ),
              ),
            ],
          ),
        const SizedBox(height: 16),
        if (_isRecording)
          PrimaryButton(
            text: 'Stop Recording',
            onPressed: _stopRecording,
            backgroundColor: Colors.red,
          ),
      ],
    );
  }

  Future<void> _startRecording() async {
    if (_controller == null || !_controller!.value.isInitialized) {
      return;
    }

    try {
      await _controller!.startVideoRecording();
      setState(() {
        _isRecording = true;
      });
    } catch (e) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('Failed to start recording: $e')),
      );
    }
  }

  Future<void> _stopRecording() async {
    if (_controller == null || !_isRecording) {
      return;
    }

    try {
      final videoFile = await _controller!.stopVideoRecording();
      setState(() {
        _isRecording = false;
      });
      widget.onVideoRecorded(videoFile.path);
    } catch (e) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('Failed to stop recording: $e')),
      );
    }
  }

  @override
  void dispose() {
    _controller?.dispose();
    super.dispose();
  }

            child: const Text('Contact Support'),
          ),
      ],
    );
  }

      case AccountStatus.active:
        return Icon(IconUtils.successIcon, color: Colors.green);
      case AccountStatus.suspended:
        return Icon(IconUtils.errorIcon, color: Colors.red);
      case AccountStatus.probation:
        return Icon(IconUtils.warningIcon, color: Colors.orange);
    }
  }

    switch (status) {
      case AccountStatus.active:
        return 'Account Active';
      case AccountStatus.suspended:
        return 'Account Suspended';
      case AccountStatus.probation:
        return 'Verification Pending';
    }
  }

    return '${date.day}/${date.month}/${date.year}';

import 'package:flutter/material.dart';
import 'package:image_picker/image_picker.dart';
import '../core/config/app_theme.dart';
import '../core/utils/theme_utils.dart';
import 'primary_button.dart';

class ImagePickerWidget extends StatefulWidget {
  final Function(XFile?) onImageSelected;
  final String? initialImageUrl;
  final String title;
  final double size;

  const ImagePickerWidget({
    super.key,
    required this.onImageSelected,
    this.initialImageUrl,
    this.title = 'Profile Photo',
    this.size = 120,
  });

  @override
  State<ImagePickerWidget> createState() => _ImagePickerWidgetState();
}

class _ImagePickerWidgetState extends State<ImagePickerWidget> {
  XFile? _selectedImage;
  String? _imageUrl;

  @override
  void initState() {
    super.initState();
    _imageUrl = widget.initialImageUrl;
  }

  Future<void> _pickImage(ImageSource source) async {
    try {
      final ImagePicker picker = ImagePicker();
      final XFile? image = await picker.pickImage(
        source: source,
        maxWidth: 800,
        maxHeight: 800,
        imageQuality: 80,
      );

      if (image != null) {
        setState(() {
          _selectedImage = image;
          _imageUrl = null;
        });
        widget.onImageSelected(image);
      }
    } catch (e) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('Failed to pick image: $e')),
      );
    }
  }

  void _showImageSourceDialog() {
    showDialog(
      context: context,
      builder: (BuildContext context) {
        return AlertDialog(
          title: Text('Choose Image Source'),
          content: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              ListTile(
                leading: Icon(Icons.camera_alt),
                title: Text('Take Photo'),
                onTap: () {
                  Navigator.pop(context);
                  _pickImage(ImageSource.camera);
                },
              ),
              ListTile(
                leading: Icon(Icons.photo_library),
                title: Text('Choose from Gallery'),
                onTap: () {
                  Navigator.pop(context);
                  _pickImage(ImageSource.gallery);
                },
              ),
            ],
          ),
        );
      },
    );
  }

  @override
  Widget build(BuildContext context) {
    final hasImage = _selectedImage != null || _imageUrl != null;

    return Column(
      children: [
        Stack(
          children: [
            Container(
              width: widget.size,
              height: widget.size,
              decoration: BoxDecoration(
                shape: BoxShape.circle,
                border: Border.all(
                  color: AppTheme.lightGrayUI,
                  width: 2,
                ),
              ),
              child: ClipOval(
                child: hasImage
                    ? _buildImage()
                    : Icon(
                        Icons.person,
                        size: widget.size * 0.6,
                        color: AppTheme.disabledGray,
                      ),
              ),
            ),
            Positioned(
              bottom: 0,
              right: 0,
              child: Container(
                width: 36,
                height: 36,
                decoration: BoxDecoration(
                  color: AppTheme.primaryRed,
                  shape: BoxShape.circle,
                  border: Border.all(
                    color: Colors.white,
                    width: 2,
                  ),
                ),
                child: IconButton(
                  icon: Icon(Icons.camera_alt, size: 18, color: Colors.white),
                  onPressed: _showImageSourceDialog,
                  padding: EdgeInsets.zero,
                ),
              ),
            ),
          ],
        ),
        const SizedBox(height: 8),
        Text(
          widget.title,
          style: Theme.of(context).textTheme.bodyMedium?.copyWith(
                fontWeight: FontWeight.w500,
              ),
        ),
      ],
    );
  }

  Widget _buildImage() {
    if (_selectedImage != null) {
      return Image.network(
        _selectedImage!.path,
        fit: BoxFit.cover,
        errorBuilder: (context, error, stackTrace) {
          return Icon(Icons.error);
        },
      );
    } else if (_imageUrl != null) {
      return Image.network(
        _imageUrl!,
        fit: BoxFit.cover,
        errorBuilder: (context, error, stackTrace) {
          return Icon(Icons.error);
        },
      );
    }
    return SizedBox();
  }
}
