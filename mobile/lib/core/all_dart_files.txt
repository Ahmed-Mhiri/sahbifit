import 'package:dio/dio.dart';
import 'package:injectable/injectable.dart';
import 'package:sahbifit/app/bloc/app_bloc.dart'; // <-- 1. Import AppBloc
import 'package:sahbifit/core/di/service_locator.dart'; // <-- 2. Import getIt
import 'package:sahbifit/core/services/secure_storage_service.dart';

@lazySingleton
class AuthInterceptor extends Interceptor {
  final SecureStorageService _storageService;

  AuthInterceptor(this._storageService);

  @override
  Future<void> onRequest(
    RequestOptions options,
    RequestInterceptorHandler handler,
  ) async {
    // Read the token from storage
    final String? token = await _storageService.readToken();

    // Add the Authorization header if the token exists
    if (token != null) {
      options.headers['Authorization'] = 'Bearer $token';
    }

    return super.onRequest(options, handler);
  }

  @override
  void onError(DioException err, ErrorInterceptorHandler handler) {
    if (err.response?.statusCode == 401) {
      print('--- 401 Unauthorized: Dispatching LogoutRequested ---');

      // 3. THIS IS THE CHANGE
      // Use getIt to find the globally registered AppBloc
      // and add the logout event. The AppBloc will handle
      // deleting the token and updating the app's state.
      try {
        getIt<AppBloc>().add(const AppEvent.logoutRequested());
      } catch (e) {
        print('Failed to get AppBloc from getIt to dispatch logout: $e');
      }
    }
    return super.onError(err, handler);
  }
}

import 'package:dio/dio.dart';

class ServerException implements Exception {
  final String message;

  ServerException({required this.message});

  /// This factory constructor creates a [ServerException] from a [DioException].
  /// It parses the Dio error to find the most helpful message.
  factory ServerException.fromDio(DioException e) {
    String message = 'An unknown server error occurred.';

    // Try to get a specific error message from the backend response
    if (e.response != null && e.response?.data != null) {
      // Check if the backend sent a map with a 'message' key
      // e.g., { "message": "Phone number is already registered." }
      if (e.response?.data is Map) {
        message = e.response?.data['message'] ?? message;
      }
      // Check if the backend just sent a plain text error
      else if (e.response?.data is String) {
        final responseString = e.response!.data as String;
        if (responseString.isNotEmpty) {
          message = responseString;
        }
      }
    } 
    // Handle common Dio error types
    else if (e.type == DioExceptionType.connectionTimeout ||
        e.type == DioExceptionType.receiveTimeout ||
        e.type == DioExceptionType.sendTimeout) {
      message = 'Connection timed out. Please try again.';
    } else if (e.type == DioExceptionType.connectionError) {
      message = 'Network error. Please check your internet connection.';
    } else if (e.type == DioExceptionType.cancel) {
      message = 'Request was cancelled.';
    }

    return ServerException(message: message);
  }
}

@lazySingleton
class PushNotificationService {
  final FirebaseMessaging _fcm = FirebaseMessaging.instance;
  final UserRepository _userRepository; // To send the token to the backend
  final FlutterLocalNotificationsPlugin _localNotifications =
      FlutterLocalNotificationsPlugin();

  PushNotificationService(this._userRepository);

  Future<void> init() async {
    // 1. Request permissions (iOS/macOS)
    await _requestPermissions();

    // 2. Initialize local notifications for foreground display
    await _initLocalNotifications();

    // 3. Get the FCM token and send it to the backend
    await _initToken();

    // 4. Set up message handlers
    _setupMessageHandlers();
  }

  Future<void> _requestPermissions() async {
    if (Platform.isIOS || Platform.isMacOS) {
      await _fcm.requestPermission(
        alert: true,
        badge: true,
        sound: true,
      );
    }
  }

  Future<void> _initToken() async {
    try {
      final String? token = await _fcm.getToken();
      if (token != null) {
        print("--- FCM Token: $token ---");
        // Send the token to your backend
        // This will call the usecase which calls the repository
        // which calls the datasource to hit PUT /api/users/me/fcm-token
        await _userRepository.updateFcmToken(token);
      }

      // Listen for token refreshes
      _fcm.onTokenRefresh.listen((newToken) async {
        print("--- FCM Token Refreshed: $newToken ---");
        await _userRepository.updateFcmToken(newToken);
      });
    } catch (e) {
      print("Error getting/sending FCM token: $e");
    }
  }

  Future<void> _initLocalNotifications() async {
    const AndroidInitializationSettings androidSettings =
        AndroidInitializationSettings('@mipmap/ic_launcher');
    const DarwinInitializationSettings iosSettings =
        DarwinInitializationSettings();

    const InitializationSettings settings = InitializationSettings(
      android: androidSettings,
      iOS: iosSettings,
    );

    await _localNotifications.initialize(
      settings,
      onDidReceiveNotificationResponse: (NotificationResponse response) {
        // Handle notification tap when app is in foreground
        if (response.payload != null && response.payload!.isNotEmpty) {
          final data = jsonDecode(response.payload!) as Map<String, dynamic>;
          _handleNotificationTap(data);
        }
      },
    );
  }

  void _setupMessageHandlers() {
    // 1. App is in the FOREGROUND
    FirebaseMessaging.onMessage.listen((RemoteMessage message) {
      print("--- FCM Message (Foreground): ${message.notification?.title}");

      final RemoteNotification? notification = message.notification;
      final AndroidNotification? android = message.notification?.android;

      // Show a local notification
      if (notification != null) {
        _localNotifications.show(
          notification.hashCode,
          notification.title,
          notification.body,
          NotificationDetails(
            android: android != null
                ? AndroidNotificationDetails(
                    'sahbifit_channel_id', // Needs setup on Android
                    'SahbiFit Notifications',
                    channelDescription: 'Main notification channel',
                    importance: Importance.max,
                    priority: Priority.high,
                    icon: android.icon,
                  )
                : null,
            iOS: const DarwinNotificationDetails(presentSound: true),
          ),
          // Pass the data payload for tap handling
          payload: jsonEncode(message.data),
        );
      }
    });

    // 2. App is in the BACKGROUND (but not terminated) and user taps notification
    FirebaseMessaging.onMessageOpenedApp.listen((RemoteMessage message) {
      print("--- FCM Message (OpenedApp): ${message.data['type']}");
      _handleNotificationTap(message.data);
    });

    // 3. App is TERMINATED and user taps notification
    _fcm.getInitialMessage().then((RemoteMessage? message) {
      if (message != null) {
        print("--- FCM Message (Initial): ${message.data['type']}");
        _handleNotificationTap(message.data);
      }
    });

    // 4. Background message handler
    FirebaseMessaging.onBackgroundMessage(_firebaseMessagingBackgroundHandler);
  }

  /// This is the main routing logic based on your backend's data payload.
  void _handleNotificationTap(Map<String, dynamic> data) {
    final String? type = data['type'];
    if (type == null) return;

    // Use getIt to find the AppRouter
    final appRouter = getIt<AppRouter>();

    switch (type) {
      case 'VERIFICATION_APPROVED':
      case 'VERIFICATION_REJECTED':
        // Navigate to the user's profile
        appRouter.push(const ProfileRoute());
        break;

      case 'PENDING_RATING':
        // This matches 'sendRatingReminder' from your backend
        final String? sessionId = data['sessionId'];
        if (sessionId != null) {
          // Navigate to the pending ratings page
          appRouter.push(const PendingRatingsRoute());
        }
        break;

      default:
        // Default action, e.g., go to home
        appRouter.push(const SessionFeedRoute());
    }
  }
}

        await RecaptchaV3.initialize(_siteKey);

  /// Returns a dummy token on mobile (as v3 is web-focused)
  /// and a real token on web.

        await init();
        if (!_isInitialized) {
          debugPrint("reCAPTCHA v3 failed to initialize.");
          throw Exception("reCAPTCHA v3 failed to initialize.");

      try {
        final String token = await RecaptchaV3.execute(action);
        debugPrint("reCAPTCHA v3 token for action '$action': $token");
        return token;
      } catch (e) {
        debugPrint("Error executing reCAPTCHA v3 action '$action': $e");
        throw Exception("Failed to get reCAPTCHA token.");
      }
    } else {
      // reCAPTCHA v3 is primarily for web.
      // For mobile, you might use Google Play Integrity or App Check.

@lazySingleton
class SecureStorageService {
  final FlutterSecureStorage _storage;

  // Use const for keys
  static const String _keyToken = 'jwt_token';

  SecureStorageService(this._storage);

  Future<void> writeToken(String token) async {
    await _storage.write(key: _keyToken, value: token);
  }

  Future<String?> readToken() async {
    return await _storage.read(key: _keyToken);
}

  Future<void> deleteToken() async {
    await _storage.delete(key: _keyToken);
  }
}

/// [Type] is the return type of the usecase (e.g., AuthToken, void).
/// [Params] is the parameters class (e.g., LoginParams).
abstract class UseCase<Type, Params> {
  /// The [call] method is the single entry point for the usecase.
  /// It's designed to be asynchronous and return either a [Failure]
  /// or the expected [Type].
  ///
  /// Note: In our Auth implementation, we are letting exceptions bubble up
  /// to the BLoC. You can also use an Either<Failure, Type> wrapper here.
  /// For simplicity, we'll stick to throwing Failures.
  Future<Type> call(Params params);
}

/// A helper class for usecases that do not require any parameters.
/// For example: `GetMyProfile(NoParams())`
class NoParams extends Equatable {
  @override
  List<Object?> get props => [];
}

/// Formats a DateTime object into the 'YYYY-MM-DD' string
/// required by your backend's `RegisterRequest`.
String formatToYyyyMmDd(DateTime date) {
  return DateFormat('yyyy-MM-dd').format(date);
}

/// Formats an ISO 8601 string (like those from OffsetDateTime)
/// into a more readable format for the UI.
///
/// Example: "2025-11-13T16:04:39+01:00" -> "Nov 13, 2025, 4:04 PM"
String formatReadableDateTime(String isoString) {
  try {
    final DateTime dateTime = DateTime.parse(isoString);
    // Format: "MMM d, yyyy, h:mm a" (e.g., "Nov 13, 2025, 4:04 PM")
    return DateFormat.yMd().add_jm().format(dateTime.toLocal());
  } catch (e) {
    print("Error formatting date: $e");
    return "Invalid Date";
  }
}

/// Formats just the time from an ISO 8601 string.
///
/// Example: "2025-11-13T16:04:39+01:00" -> "4:04 PM"
String formatReadableTime(String isoString) {
   try {
    final DateTime dateTime = DateTime.parse(isoString);
    // Format: "h:mm a" (e.g., "4:04 PM")
    return DateFormat.jm().format(dateTime.toLocal());
  } catch (e) {
    print("Error formatting time: $e");
    return "--:--";
  }
}

/// Validates a phone number based on the backend's Tunisian number regex.
///
/// Regex: `^(2|3|4|5|7|9)\d{7}$`
/// (Must start with 2, 3, 4, 5, 7, or 9, followed by 7 digits).
String? validateTunisianPhoneNumber(String? value) {
  if (value == null || value.isEmpty) {
    return 'Phone number cannot be empty.';
  }
  
  // This regex matches your Java backend's logic
  final RegExp tunisianPhoneRegex = RegExp(r'^(2|3|4|5|7|9)\d{7}$');
  
  if (!tunisianPhoneRegex.hasMatch(value)) {
    return 'Invalid Tunisian phone number (8 digits, no +216).';
  }
  
  return null; // Valid
}

import 'package:dartz/dartz.dart';

/// A type definition for a Future that returns either a Failure or a Type [T].
/// This is commonly used in repositories and use cases.
///
/// Example: `FutureEither<User> getMyProfile()`
typedef FutureEither<T> = Future<Either<Failure, T>>;

/// A type definition for a Future that returns either a Failure or void.
/// Used for operations that don't return data, like `deleteToken()`.
///
/// Example: `FutureVoid deleteToken()`
typedef FutureVoid = FutureEither<void>;

/// A common type definition for JSON data.
///
/// Example: `DataMap json = response.data as DataMap;`
typedef DataMap = Map<String, dynamic>;

    return '$fieldName cannot be empty.';
  }
  return null; // Valid
}

/// Validates the password based on your backend's `RegisterRequest`.
/// (`@Size(min = 8, message = "Password must be at least 8 characters long.")`)
String? validatePassword(String? value) {
  if (value == null || value.isEmpty) {
    return 'Password cannot be empty.';
  }
  if (value.length < 8) {
    return 'Password must be at least 8 characters long.';
  }
  return null; // Valid
}

/// Validates the OTP code based on your backend's `VerifySmsRequest`.
/// (`@Size(min = 6, max = 6, message = "OTP must be 6 digits long.")`)
String? validateOtp(String? value) {
  if (value == null || value.isEmpty) {
    return 'OTP cannot be empty.';
  }
  if (value.length != 6) {
    return 'OTP must be 6 digits long.';
  }
  // Optional: Check if all characters are digits
  if (int.tryParse(value) == null) {
    return 'OTP must only contain numbers.';
  }
  return null; // Valid
}
