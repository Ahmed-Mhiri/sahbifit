abstract class VerificationRemoteDatasource {
  Future<VerificationStatusModel> getVerificationStatus();
  Future<void> submitVerificationVideo(File video);
}

@Injectable(as: VerificationRemoteDatasource)
class VerificationRemoteDatasourceImpl implements VerificationRemoteDatasource {
  final ApiClient _apiClient;

  VerificationRemoteDatasourceImpl(this._apiClient);

  @override
  Future<VerificationStatusModel> getVerificationStatus() async {
    try {
      final response = await _apiClient.dio.get(
        ApiEndpoints.verificationStatus,
      );
      return VerificationStatusModel.fromJson(response.data);
    } on DioException catch (e) {
      throw ServerException.fromDio(e);
    }
  }

  @override
  Future<void> submitVerificationVideo(File video) async {
    try {
      // 1. Get the file name
      String fileName = p.basename(video.path);

      // 2. Create the FormData object
      // This is the multipart request body
      FormData formData = FormData.fromMap({
        // The key "video" MUST match the @RequestParam("video")
        // in your VerificationController.java
        "video": await MultipartFile.fromFile(
          video.path,
          filename: fileName,
        ),
      });

      // 3. Send the request
      await _apiClient.dio.post(
        ApiEndpoints.submitVerification,
        data: formData,
        // Optional: Show upload progress
        onSendProgress: (int sent, int total) {
          print('Upload progress: ${((sent / total) * 100).toStringAsFixed(0)}%');
        },
      );
      // Backend returns a 200 OK with a simple string, which we ignore
    } on DioException catch (e) {
      throw ServerException.fromDio(e);
    }
  }
}

@injectable
class VerificationCubit extends Cubit<VerificationState> {
  final GetVerificationStatus _getVerificationStatus;
  final SubmitVerificationVideo _submitVerificationVideo;

  VerificationCubit(
    this._getVerificationStatus,
    this._submitVerificationVideo,
  ) : super(const VerificationState.initial());

  /// Fetches the user's current verification status from the server.
  /// This is called from main.dart on app start.
  Future<void> fetchStatus() async {
    emit(const VerificationState.loading());
    try {
      final status = await _getVerificationStatus(NoParams());
      emit(VerificationState.loaded(status));
    } on ServerFailure catch (e) {
      emit(VerificationState.error(e.message));
    } catch (e) {
      emit(VerificationState.error(e.toString()));
    }
  }

  /// Submits the verification video.
  /// On success, it immediately re-fetches the status to update
  /// the state from 'not_started' to 'pending_review'.
  Future<void> submitVideo(File video) async {
    emit(const VerificationState.loading());
    try {
      final params = SubmitVerificationVideoParams(video: video);
      await _submitVerificationVideo(params);

      // On success, refresh the status
      await fetchStatus();
    } on ServerFailure catch (e) {
      emit(VerificationState.error(e.message));
    } catch (e) {
      emit(VerificationState.error(e.toString()));
    }
  }
}
