import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:injectable/injectable.dart';
import 'package:sahbifit/features/chat/data/models/message_model.dart';

abstract class ChatRemoteDatasource {
  Stream<List<MessageModel>> getChatStream(String sessionId);
  Future<void> sendMessage(String sessionId, MessageModel message);
}

@Injectable(as: ChatRemoteDatasource)
class ChatRemoteDatasourceImpl implements ChatRemoteDatasource {
  final FirebaseFirestore _firestore;

  ChatRemoteDatasourceImpl(this._firestore);

  // Your backend's FirebaseChatService.java creates docs in "session_chats"
  static const String _collection = 'session_chats';
  static const String _subCollection = 'messages';

  @override
  Stream<List<MessageModel>> getChatStream(String sessionId) {
    // 1. Get the collection of messages for the given session
    final collectionRef = _firestore
        .collection(_collection)
        .doc(sessionId)
        .collection(_subCollection)
        .orderBy('timestamp', descending: true); // Order by time

    // 2. Get a stream of snapshots
    return collectionRef.snapshots().map((querySnapshot) {
      // 3. For each snapshot, map the documents into MessageModels
      return querySnapshot.docs
          .map((doc) => MessageModel.fromSnapshot(doc))
          .toList();
    });
  }

  @override
  Future<void> sendMessage(String sessionId, MessageModel message) async {
    // 1. Get the reference to the "messages" subcollection
    final collectionRef = _firestore
        .collection(_collection)
        .doc(sessionId)
        .collection(_subCollection);

    // 2. Add the new message
    // We use .toFirestore() to convert the model to a map
    // and add the server-side timestamp.
    await collectionRef.add(message.toFirestore());
  }
}

class MessageModel with _$MessageModel {
  const MessageModel._();

  @JsonSerializable(
    fieldRename: FieldRename.snake, // Firestore often uses snake_case
    converters: [TimestampConverter()], // Use our custom converter
  )
  const factory MessageModel({
    @JsonKey(ignore: true) String? id, // We'll get the ID from the doc snapshot
    required String senderId,
    required String text,
    // When reading, 'timestamp' will be a Timestamp.
    // When writing, this will be null and replaced by toFirestore
    required DateTime? timestamp,
  }) = _MessageModel;

  /// Special factory for creating a model to *send* to Firestore
  /// This sets the timestamp to null, so we can use a server timestamp.
  factory MessageModel.forSending({
    required String senderId,
    required String text,
  }) =>
      MessageModel(
        senderId: senderId,
        text: text,
        timestamp: null, // Will be replaced by server timestamp
      );

  /// Factory for creating a model from a Firestore snapshot
  factory MessageModel.fromSnapshot(DocumentSnapshot doc) {
    final data = doc.data() as Map<String, dynamic>;
    // Use the generated fromJson factory, then add the doc ID
    return MessageModel.fromJson(data).copyWith(id: doc.id);
  }

  /// Factory for creating a model from a JSON map (required by freezed)
  /// This is the line that fixes the error.
  factory MessageModel.fromJson(Map<String, dynamic> json) =>
      _$MessageModelFromJson(json);

  /// Converts this model into a Map to *write* to Firestore
  Map<String, dynamic> toFirestore() {
    // toJson() will now exist thanks to the .fromJson factory being added
    // and build_runner being run.
    final json = toJson();

    // Replace our null timestamp with the server's command
    json['timestamp'] = FieldValue.serverTimestamp();
    
    // Don't write the 'id' field back into the document data
    json.remove('id'); 
    
    return json;
  }

  // Converts the data-layer model to a domain-layer entity
  Message toEntity() {
    return Message(
      messageId: id ?? '', // Use the ID from the snapshot
      senderId: senderId,
      text: text,
      timestamp: timestamp ?? DateTime.now(), // Fallback if timestamp is null
    );
  }
}

// This is the custom converter for handling Firestore Timestamps
class TimestampConverter implements JsonConverter<DateTime?, Object> {
  const TimestampConverter();

  @override
  DateTime? fromJson(Object json) {
    if (json is Timestamp) {
      return json.toDate();
    }
    // Handle cases where it might already be null
    if (json == null) {
      return null;
    }
    // You could add more robust parsing here if needed, e.g., from String
    return null;
  }

  @override
  Object toJson(DateTime? time) {
    // If a time is provided, convert it to a Timestamp
    if (time != null) {
      return Timestamp.fromDate(time);
    }
    // If time is null, we intend to use a server timestamp.
    // The `toFirestore` method handles this explicitly, but setting
    // it here ensures consistency if toJson() were called directly.
    return FieldValue.serverTimestamp();
  }
}

  ChatRepositoryImpl(this._remoteDatasource);

  @override
  Stream<List<Message>> getChatStream(String sessionId) {
    // We catch errors on the stream itself
    return _remoteDatasource.getChatStream(sessionId).map((models) {
      return models.map((model) => model.toEntity()).toList();
    }).handleError((error) {
      // Log the error and bubble up a generic failure
      print('Firestore Stream Error: $error');
      throw ServerFailure(message: 'Could not load chat messages.');
    });
  }

  @override
  Future<void> sendMessage({
    required String sessionId,
    required String senderId,
    required String text,
  }) async {
    try {
      // Create the model to send
      final message = MessageModel.forSending(
        senderId: senderId,
        text: text,
      );
      await _remoteDatasource.sendMessage(sessionId, message);
    } on FirebaseException catch (e) {
      // Handle specific Firestore errors
      throw ServerFailure(message: e.message ?? 'Failed to send message.');
    } catch (e) {
      throw ServerFailure(message: 'An unknown error occurred.');
    }
  }

@injectable
class ChatCubit extends Cubit<ChatState> {
  final GetChatStream _getChatStream;
  final SendMessage _sendMessage;

  // This subscription needs to be stored so we can cancel it later.
  StreamSubscription<List<Message>>? _messageSubscription;

  ChatCubit(
    this._getChatStream,
    this._sendMessage,
  ) : super(const ChatState.initial());

  /// Connects to the real-time message stream for a specific session.
  Future<void> listenToChat(String sessionId) async {
    // Cancel any previous subscription before starting a new one
    await _messageSubscription?.cancel();
    emit(const ChatState.loading());

    try {
      // 1. Call the use case to get the stream
      final messageStream = await _getChatStream(
        ChatStreamParams(sessionId: sessionId),
      );

      // 2. Listen to the stream
      _messageSubscription = messageStream.listen(
        (messages) {
          // 3. Every time new data comes in, emit a 'loaded' state
          emit(ChatState.loaded(messages: messages));
        },
        onError: (error) {
          // 4. Handle any errors that occur on the stream
          emit(ChatState.error(error.toString()));
        },
      );
    } on ServerFailure catch (e) {
      emit(ChatState.error(e.message));
    } catch (e) {
      emit(ChatState.error(e.toString()));
    }
  }

  /// Sends a new chat message.
  Future<void> sendMessage({
    required String sessionId,
    required String senderId, // This should be the ID of the logged-in user
    required String text,
  }) async {
    // Don't emit a loading state here.
    // The UI will update optimistically when the stream listener
    // receives the new message from Firestore.
    try {
      final params = SendMessageParams(
        sessionId: sessionId,
        senderId: senderId,
        text: text,
      );
      await _sendMessage(params);
      // On success, do nothing. The stream listener will update the state.
    } on ServerFailure catch (e) {
      // If sending fails, emit an error state
      emit(ChatState.error(e.message));
    } catch (e) {
