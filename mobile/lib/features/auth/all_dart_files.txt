// This is the abstract interface
abstract class AuthRemoteDatasource {
  Future<AuthResponseModel> login(LoginRequestModel request);
  Future<void> register(RegisterRequestModel request);
  Future<void> verifySms(VerifySmsRequestModel request);
  Future<void> resendOtp(ResendOtpRequestModel request);

// This is the implementation
@Injectable(as: AuthRemoteDatasource)
class AuthRemoteDatasourceImpl implements AuthRemoteDatasource {
  final ApiClient _apiClient;

  AuthRemoteDatasourceImpl(this._apiClient);

  @override
  Future<AuthResponseModel> login(LoginRequestModel request) async {
    try {
      final response = await _apiClient.dio.post(
        '/auth/login',
        data: request.toJson(),
      );
      // Dio successfully gets a 2xx response
      return AuthResponseModel.fromJson(response.data);
    } on DioException catch (e) {
      // Re-throw a custom exception for the repository layer to catch
      throw ServerException.fromDio(e);
    }
  }

  @override
  Future<void> register(RegisterRequestModel request) async {
    try {
      await _apiClient.dio.post(
        '/auth/register',
        data: request.toJson(),
      );

      throw ServerException.fromDio(e);

        '/auth/verify-sms',
        data: request.toJson(),
      );
    } on DioException catch (e) {
      throw ServerException.fromDio(e);
    }
