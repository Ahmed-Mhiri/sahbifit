import 'package:dio/dio.dart'; // Assuming you use Dio for HTTP
import 'package:sahbifit_mobile/features/gym/data/models/gym_model.dart';
import 'package:sahbifit_mobile/features/gym/data/models/gym_submission_model.dart';
import 'package:sahbifit_mobile/core/errors/exceptions.dart';
abstract class GymRemoteDataSource {
  /// Calls the GET /api/gyms endpoint.
  /// Throws a [ServerException] for all error codes.
  Future<List<GymModel>> getApprovedGyms();

  /// Calls the POST /api/gyms/submit endpoint.
  /// Throws a [ServerException] for all error codes.
  Future<void> submitGym(GymSubmissionModel submissionModel);
}

class GymRemoteDataSourceImpl implements GymRemoteDataSource {
  final Dio dio;

  GymRemoteDataSourceImpl({required this.dio});

  @override
  Future<List<GymModel>> getApprovedGyms() async {
    try {
      // This path matches GymController's @GetMapping("/gyms")
      final response = await dio.get('/api/gyms');

      if (response.statusCode == 200) {
        final List<dynamic> gymsJson = response.data as List;
        return gymsJson
            .map((json) => GymModel.fromJson(json as Map<String, dynamic>))
            .toList();
      } else {
        throw ServerException();
      }
    } catch (e) {
      throw ServerException();
    }
  }

  @override
  Future<void> submitGym(GymSubmissionModel submissionModel) async {
    try {
      // This path matches GymController's @PostMapping("/gyms/submit")
      final response = await dio.post(
        '/api/gyms/submit',
        data: submissionModel.toJson(),
      );

      // HttpStatus.CREATED is 201
      if (response.statusCode != 201) {
        throw ServerException();
      }
      // No return value needed, as per controller's ResponseEntity<Gym>
      // We just care that it succeeded.
    } catch (e) {
      throw ServerException();
    }
  }
}

import 'package:equatable/equatable.dart';
import 'package:sahbifit_mobile/features/gym/domain/enums/gym_status.dart';
import '../../domain/entities/gym.dart'; // We will create this

class GymModel extends Gym {
  const GymModel({
    required super.id,
    required super.name,
    required super.address,
    required super.memberCount,
    required this.status, // The model has status, the entity might not
  });

  final GymStatus status;

  factory GymModel.fromJson(Map<String, dynamic> json) {
    return GymModel(
      // Matches gymId from Gym.java
      id: json['gymId'] as String, 
      
      // Matches name from Gym.java
      name: json['name'] as String, 
      
      // Matches address from Gym.java
      address: json['address'] as String? ?? '', 
      
      // Matches @Formula memberCount from Gym.java
      // This directly implements MVP 6.2
      memberCount: json['memberCount'] as int, 
      
      // Matches status from Gym.java
      status: (json['status'] as String).toGymStatus(),
    );
  }

  // Helper to convert string to enum
}

extension on String {
  GymStatus toGymStatus() {
    switch (this) {
      case 'pending':
        return GymStatus.pending;
      case 'approved':
        return GymStatus.approved;
      case 'rejected':
        return GymStatus.rejected;
      default:
        return GymStatus.pending; // Default fallback
    }
  }
}

import 'package:flutter_bloc/flutter_bloc.dart';
import 'package:sahbifit_mobile/features/gym/presentation/cubit/gym_list_state.dart';
import '../../../../../core/usecases/usecase.dart'; // For NoParams
import '../../domain/usecases/get_approved_gyms.dart';

class GymListCubit extends Cubit<GymListState> {
  final GetApprovedGyms getApprovedGyms;

  GymListCubit({required this.getApprovedGyms}) : super(GymListInitial());

  Future<void> loadGyms() async {
    emit(GymListLoading());
    
    final failureOrGyms = await getApprovedGyms(NoParams());
    
    failureOrGyms.fold(
      (failure) => emit(GymListError(message: 'Failed to load gyms')),
      (gyms) => emit(GymListLoaded(gyms: gyms)),
    );
  }
}
