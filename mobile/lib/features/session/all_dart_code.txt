import 'package:dartz/dartz.dart';
import 'package:dio/dio.dart'; // Assuming you use Dio
import 'package:sahbifit_mobile/core/errors/failure.dart';
import 'package:sahbifit_mobile/core/utils/type_defs.dart';
import 'package:sahbifit_mobile/features/session/data/datasources/session_remote_datasource.dart';
import 'package:sahbifit_mobile/features/session/data/models/create_session_model.dart';
import 'package:sahbifit_mobile/features/session/data/models/mark_attendance_model.dart';
import 'package:sahbifit_mobile/features/session/data/models/session_model.dart';
import 'package:sahbifit_mobile/features/session/domain/entities/session.dart';
import 'package:sahbifit_mobile/features/session/domain/repositories/session_repository.dart';
import 'package:sahbifit_mobile/features/verification/domain/enums/verified_gender_enum.dart';

class SessionRepositoryImpl implements SessionRepository {
  final SessionRemoteDataSource remoteDataSource;

  SessionRepositoryImpl(this.remoteDataSource);

  @override
  Future<Result<List<Session>>> getActiveSessions(
      {VerifiedGender? genderFilter}) async {
    try {
      final sessionModels =
          await remoteDataSource.getActiveSessions(genderFilter: genderFilter);
      // Map from Data Models to Domain Entities
      final sessions = sessionModels.map((model) => model.toDomain()).toList();
      return Right(sessions);
    } on DioException catch (e) {
      // Handle API errors
      return Left(ServerFailure(e.message ?? 'Failed to fetch sessions'));
    } catch (e) {
      return Left(ServerFailure(e.toString()));
    }
  }

  @override
  Future<Result<Session>> createSession(
      CreateSessionModel createSessionModel) async {
    try {
      final sessionModel =
          await remoteDataSource.createSession(createSessionModel);
      // Map from Data Model to Domain Entity
      return Right(sessionModel.toDomain());
    } on DioException catch (e) {
      // Handle specific backend errors (like "IllegalStateException")
      if (e.response?.statusCode == 400 || e.response?.statusCode == 500) {
        // You can parse the error response from Spring Boot
        final errorMessage =
            e.response?.data['message'] ?? 'Failed to create session.';
        return Left(ServerFailure(errorMessage));
      }
      return Left(ServerFailure(e.message ?? 'Failed to create session'));
    } catch (e) {
      return Left(ServerFailure(e.toString()));
    }
  }

  @override
  Future<Result<void>> joinSession(String sessionId) async {
    try {
      await remoteDataSource.joinSession(sessionId);
      return const Right(null);
    } on DioException catch (e) {
      // Handle specific backend errors
      if (e.response?.statusCode == 400 || e.response?.statusCode == 500) {
        final errorMessage =
            e.response?.data['message'] ?? 'Failed to join session.';
        return Left(ServerFailure(errorMessage));
      }
      return Left(ServerFailure(e.message ?? 'Failed to join session'));
    } catch (e) {
      return Left(ServerFailure(e.toString()));
    }
  }

  @override
  Future<Result<void>> markAttendance(
      MarkAttendanceModel markAttendanceModel) async {
    try {
      await remoteDataSource.markAttendance(markAttendanceModel);
      return const Right(null);
    } on DioException catch (e) {
      if (e.response?.statusCode == 403) {
        return const Left(ServerFailure('You are not allowed to mark attendance.'));
      }
      return Left(ServerFailure(e.message ?? 'Failed to mark attendance'));
    } catch (e) {
      return Left(ServerFailure(e.toString()));
    }
  }
}

import 'package:flutter_bloc/flutter_bloc.dart';
import 'package:sahbifit_mobile/features/session/domain/usecases/get_active_sessions.dart';
import 'package:sahbifit_mobile/features/session/presentation/cubit/session_feed_state.dart';
import 'package:sahbifit_mobile/features/verification/domain/enums/verified_gender_enum.dart';

class SessionFeedCubit extends Cubit<SessionFeedState> {
  final GetActiveSessions _getActiveSessions;

  SessionFeedCubit(this._getActiveSessions)
      : super(const SessionFeedState.initial());

  /// Fetches the active sessions from the repository.
  Future<void> fetchActiveSessions({VerifiedGender? genderFilter}) async {
    emit(const SessionFeedState.loading());

    final params = GetActiveSessionsParams(genderFilter: genderFilter);
    final result = await _getActiveSessions(params);

    result.fold(
      // On Failure
      (failure) => emit(SessionFeedState.failure(failure.message)),
      // On Success
      (sessions) => emit(SessionFeedState.success(sessions)),
    );
  }
}

    required this.sessionsCompleted,

    final sessionsRemaining = sessionsRequired - sessionsCompleted;

    return AlertDialog(
      title: Row(
        children: [

          Text('Build Your Reputation'),
        ],

      content: Column(
        mainAxisSize: MainAxisSize.min,

          Text(

            style: Theme.of(context).textTheme.bodyMedium,
          ),
          const SizedBox(height: 16),
          _buildProgressIndicator(context),
          const SizedBox(height: 16),
          Text(
            'Complete $sessionsRemaining more group sessions to unlock this feature.',
            style: Theme.of(context).textTheme.bodySmall?.copyWith(
                  fontStyle: FontStyle.italic,
                ),
          ),
        ],
      ),
      actions: [
        TextButton(
          onPressed: onClose,
          child: Text('Close'),
        ),
        PrimaryButton(
          text: 'Browse Group Sessions',
          onPressed: () {
            onClose();
            // Navigate to group sessions
          },

    final progress = sessionsCompleted / sessionsRequired;

        Text(

          style: Theme.of(context).textTheme.bodySmall,

          value: progress,
          backgroundColor: Colors.grey[300],

import 'package:flutter/material.dart';
import '../../../../../shared_widgets/theme_components.dart';
import '../../../../../core/utils/icon_utils.dart';

class SessionParticipantList extends StatelessWidget {
  final List<SessionParticipant> participants;
  final bool showAttendanceActions;
  final Function(String, bool)? onAttendanceChanged;

  const SessionParticipantList({
    super.key,
    required this.participants,
    this.showAttendanceActions = false,
    this.onAttendanceChanged,
  });

  @override
  Widget build(BuildContext context) {
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Text(
          'Participants (${participants.length})',
          style: Theme.of(context).textTheme.titleMedium,
        ),
        const SizedBox(height: 16),
        if (participants.isEmpty)
          const Text('No participants yet')
        else
          ...participants.map((participant) => _buildParticipantCard(context, participant)),
      ],
    );
  }

  Widget _buildParticipantCard(BuildContext context, SessionParticipant participant) {
    return SahbifitCard(
      padding: const EdgeInsets.all(12),
      margin: const EdgeInsets.only(bottom: 8),
      child: Row(
        children: [
          CircleAvatar(
            radius: 24,
            backgroundImage: participant.avatarUrl != null
                ? NetworkImage(participant.avatarUrl!)
                : null,
            child: participant.avatarUrl == null
                ? const Icon(Icons.person)
                : null,
          ),
          const SizedBox(width: 12),
          Expanded(
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                Row(
                  children: [
                    Text(
                      participant.name,
                      style: Theme.of(context).textTheme.bodyLarge?.copyWith(
                            fontWeight: FontWeight.w600,
                          ),
                    ),
                    if (participant.isVerified) ...[
                      const SizedBox(width: 4),
                      Icon(IconUtils.verifiedBadge, size: 16, color: Colors.blue),
                    ],
                  ],
                ),
                Text(
                  participant.status.displayName,
                  style: Theme.of(context).textTheme.bodySmall?.copyWith(
                        color: _getStatusColor(participant.status),
                      ),
                ),
              ],
            ),
          ),
          if (showAttendanceActions && onAttendanceChanged != null)
            _buildAttendanceToggle(participant),
        ],
      ),
    );
  }

  Widget _buildAttendanceToggle(SessionParticipant participant) {
    final isPresent = participant.status == ParticipantStatus.attended;
    
    return Row(
      mainAxisSize: MainAxisSize.min,
      children: [
        Text(
          isPresent ? 'Present' : 'Absent',
          style: TextStyle(
            color: isPresent ? Colors.green : Colors.red,
            fontWeight: FontWeight.w500,
          ),
        ),
        const SizedBox(width: 8),
        Switch(
          value: isPresent,
          onChanged: (value) {
            onAttendanceChanged!(participant.id, value);
          },
          activeColor: Colors.green,
        ),
      ],
    );
  }

  Color _getStatusColor(ParticipantStatus status) {
    switch (status) {
      case ParticipantStatus.attended:
        return Colors.green;
      case ParticipantStatus.registered:
        return Colors.blue;
      case ParticipantStatus.noShow:
        return Colors.red;
      case ParticipantStatus.pending:
        return Colors.orange;
    }
  }
}

import 'package:flutter/material.dart';
import 'package:sahbifit_mobile/core/config/app_theme.dart';
import '../../../../core/utils/theme_utils.dart';
import '../../../../core/utils/icon_utils.dart';

class SessionTypeToggle extends StatelessWidget {
  final bool isGroupSession;
  final ValueChanged<bool> onChanged;
  final bool isEnabled;

  const SessionTypeToggle({
    super.key,
    required this.isGroupSession,
    required this.onChanged,
    this.isEnabled = true,
  });

  @override
  Widget build(BuildContext context) {
    return Container(
      padding: const EdgeInsets.all(4),
      decoration: BoxDecoration(
        color: AppTheme.lightGrayUI,
        borderRadius: BorderRadius.circular(12),
      ),
      child: Row(
        children: [
          _buildToggleOption(
            context,
            isSelected: isGroupSession,
            isGroup: true,
            onTap: isEnabled ? () => onChanged(true) : null,
          ),
          _buildToggleOption(
            context,
            isSelected: !isGroupSession,
            isGroup: false,
            onTap: isEnabled ? () => onChanged(false) : null,
          ),
        ],
      ),
    );
  }

  Widget _buildToggleOption(
    BuildContext context, {
    required bool isSelected,
    required bool isGroup,
    required VoidCallback? onTap,
  }) {
    final label = isGroup ? 'Group Session' : '1-on-1 Session';
    final icon = isGroup ? IconUtils.groupSessionIcon : IconUtils.oneOnOneSessionIcon;
    
    return Expanded(
      child: InkWell(
        onTap: onTap,
        borderRadius: BorderRadius.circular(8),
        child: Container(
          padding: const EdgeInsets.symmetric(vertical: 12, horizontal: 8),
          decoration: BoxDecoration(
            color: isSelected ? Colors.white : Colors.transparent,
            borderRadius: BorderRadius.circular(8),
            boxShadow: isSelected
                ? [
                    BoxShadow(
                      color: Colors.black.withOpacity(0.1),
                      blurRadius: 4,
                      offset: const Offset(0, 2),
                    ),
                  ]
                : null,
          ),
          child: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              Icon(
                icon,
                color: isSelected
                    ? ThemeUtils.getSessionTypeColor(isGroup)
                    : AppTheme.disabledGray,
                size: 20,
              ),
              const SizedBox(height: 4),
              Text(
                label,
                style: Theme.of(context).textTheme.bodySmall?.copyWith(
                      fontWeight: isSelected ? FontWeight.w600 : FontWeight.normal,
                      color: isSelected ? AppTheme.primaryDarkGray : AppTheme.disabledGray,
                    ),
                textAlign: TextAlign.center,
              ),
            ],
          ),
        ),
      ),
    );
  }
}
