  // --- ADDED ---
  final FirebaseFirestore _firestore;

  // --- MODIFIED CONSTRUCTOR ---
  SessionRemoteDatasourceImpl(this._apiClient, this._firestore);

  @override
  Future<List<SessionModel>> getActiveSessions(
      VerifiedGender? genderFilter) async {
    try {
      final Map<String, dynamic> queryParams = {};
      if (genderFilter != null) {
        queryParams['genderFilter'] = genderFilter.name;
      }

      final response = await _apiClient.dio.get(
        ApiEndpoints.sessions,
        queryParameters: queryParams.isNotEmpty ? queryParams : null,
      );
      return (response.data as List)
          .map((json) => SessionModel.fromJson(json))
          .toList();
    } on DioException catch (e) {
      throw ServerException.fromDio(e);
    }
  }

  @override
  Future<SessionModel> createSession(CreateSessionModel request) async {
    try {
      final response = await _apiClient.dio.post(
        ApiEndpoints.createSession,
        data: request.toJson(),
      );
      // Backend returns the created session
      return SessionModel.fromJson(response.data);
    } on DioException catch (e) {
      throw ServerException.fromDio(e);
    }
  }

  @override
  Future<SessionModel> getSessionById(String sessionId) async {
    try {
      // Use the base 'sessions' endpoint and append the ID
      final response = await _apiClient.dio.get(
        '${ApiEndpoints.sessions}/$sessionId',
      );
      return SessionModel.fromJson(response.data);
    } on DioException catch (e) {
      throw ServerException.fromDio(e);
    }
  }

  @override
  Future<void> joinSession(String sessionId) async {
    try {
      await _apiClient.dio.post(
        ApiEndpoints.joinSession(sessionId), // Use function for dynamic URL
      );
    } on DioException catch (e) {
      throw ServerException.fromDio(e);
    }
  }

  @override
  Future<void> markAttendance(MarkAttendanceModel request) async {
    try {
      await _apiClient.dio.put(
        ApiEndpoints.markAttendance,
        data: request.toJson(),
      );
    } on DioException catch (e) {
      throw ServerException.fromDio(e);
    }
  }

  // --- ADDED ---
  @override
  Stream<List<SessionParticipantModel>> getParticipantsStream(
      String sessionId) {
    // No try-catch needed here; the stream itself will propagate errors.
    return _firestore
        .collection('sessions')
        .doc(sessionId)
        .collection('participants')
        .snapshots()
        .map((snapshot) {
      return snapshot.docs
          .map((doc) => SessionParticipantModel.fromJson(doc.data()))
          .toList();
    });
  }
}

@injectable
class CreateSessionCubit extends Cubit<CreateSessionState> {
  final CreateSession _createSession;

  CreateSessionCubit(this._createSession)
      : super(CreateSessionState.initial(
          // Default to 1 hour from now, duration 1 hour
          startTime: DateTime.now().add(const Duration(hours: 1)),
          endTime: DateTime.now().add(const Duration(hours: 2)),
        ));

  void titleChanged(String title) {
    emit(state.copyWith(title: title, status: CreateSessionStatus.initial, errorMessage: null));
  }

  void gymChanged(Gym gym) {
    emit(state.copyWith(selectedGym: gym, status: CreateSessionStatus.initial, errorMessage: null));
  }

  void sessionTypeChanged(SessionType type) {
    emit(state.copyWith(sessionType: type, status: CreateSessionStatus.initial, errorMessage: null));
  }

  void startTimeChanged(DateTime time) {
    emit(state.copyWith(startTime: time, status: CreateSessionStatus.initial, errorMessage: null));
  }

  void endTimeChanged(DateTime time) {
    emit(state.copyWith(endTime: time, status: CreateSessionStatus.initial, errorMessage: null));
  }

  void maxParticipantsChanged(int count) {
    emit(state.copyWith(maxParticipants: count, status: CreateSessionStatus.initial, errorMessage: null));
  }

  Future<void> submitCreateSession() async {
    if (state.selectedGym == null || state.title.trim().isEmpty) {
      emit(state.copyWith(
          status: CreateSessionStatus.error,
          errorMessage: 'Title and Gym are required.'));
      return;
    }

    if (state.endTime.isBefore(state.startTime.add(const Duration(minutes: 30)))) {
       emit(state.copyWith(
          status: CreateSessionStatus.error,
          errorMessage: 'End time must be at least 30 minutes after start time.'));
      return;
    }

    emit(state.copyWith(status: CreateSessionStatus.loading, errorMessage: null));
    try {
      final params = CreateSessionParams(
        gymId: state.selectedGym!.gymId,
        sessionType: state.sessionType,
        title: state.title.trim(),
        startTime: state.startTime,
        endTime: state.endTime,
        maxParticipants: state.maxParticipants,
      );
      final session = await _createSession(params);
      emit(state.copyWith(
          status: CreateSessionStatus.success, createdSession: session));
    } on ServerFailure catch (e) {
      emit(
          state.copyWith(status: CreateSessionStatus.error, errorMessage: e.message));
    } catch(e) {
       emit(
          state.copyWith(status: CreateSessionStatus.error, errorMessage: e.toString()));
    }
  }
}

class SessionFeedCubit extends Cubit<SessionFeedState> {
  final GetActiveSessions _getActiveSessions;

  SessionFeedCubit(this._getActiveSessions)
      : super(const SessionFeedState.initial());

  /// Fetches the list of active sessions, optionally applying a gender filter.
  Future<void> fetchSessions({VerifiedGender? genderFilter}) async {
    emit(const SessionFeedState.loading());
    try {
      final params = GetActiveSessionsParams(genderFilter: genderFilter);
      final sessions = await _getActiveSessions(params);

      emit(SessionFeedState.loaded(
        sessions: sessions,
        genderFilter: genderFilter,
      ));
    } on ServerFailure catch (e) {
      emit(SessionFeedState.error(e.message));
    } catch (e) {
      emit(SessionFeedState.error(e.toString()));
    }
  }
}

      physics: const NeverScrollableScrollPhysics(),
      itemCount: participants.length,
      itemBuilder: (context, index) {
        final participant = participants[index];
        return ListTile(
          leading: CircleAvatar(
            // Placeholder for profile pic
            backgroundColor: AppTheme.lightGray,
            child: Text(participant.user.firstName.substring(0, 1)),
          ),
          title: Text(
            participant.user.firstName,
            style: context.textTheme.titleMedium,
          ),
          subtitle: Text(
            'Status: ${participant.status.name}',
            style: context.textTheme.bodySmall?.copyWith(
              color: _getStatusColor(participant.status),
            ),
          ),
          trailing:
              onMarkAttendance != null ? _buildAttendanceButtons(context, participant) : null,
        );
      },
    );
  }

  Widget _buildAttendanceButtons(BuildContext context, SessionParticipant participant) {
    return Row(
      mainAxisSize: MainAxisSize.min,
      children: [
        IconButton(
          icon: Icon(
            Icons.check_circle_rounded,
            color: participant.status == ParticipantStatus.attended
                ? AppTheme.successGreen
                : AppTheme.disabledGray,
          ),
          onPressed: () =>
              onMarkAttendance!(participant.userId, ParticipantStatus.attended),
        ),
        IconButton(
          icon: Icon(
            Icons.cancel_rounded,
            color: participant.status == ParticipantStatus.noShow
                ? AppTheme.primaryRed
                : AppTheme.disabledGray,
          ),
          onPressed: () =>
              onMarkAttendance!(participant.userId, ParticipantStatus.noShow),
        ),
      ],
    );
  }

  Color _getStatusColor(ParticipantStatus status) {
    switch (status) {
      case ParticipantStatus.attended:
        return AppTheme.successGreen;
      case ParticipantStatus.noShow:
        return AppTheme.primaryRed;
      case ParticipantStatus.joined:
        return AppTheme.trustBlue;
      default:
        return AppTheme.disabledGray;
    }
  }
}
