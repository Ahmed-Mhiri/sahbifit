@Slf4j
@Service
public class AdminService {

    @Autowired
    private UserRepository userRepository;

    @Autowired
    private StorageService storageService; // For file deletion

    @Autowired
    private ReportRepository reportRepository;

    @Autowired
    private PushNotificationService pushNotificationService;

    /**
     * Retrieves all users waiting for verification.
     */
    public List<VerificationQueueDto> getVerificationQueue() {
        List<User> pendingUsers = userRepository.findByVerificationStatus(VerificationStatus.pending_review);

        return pendingUsers.stream().map(user -> {
            // CHANGED: Using record constructor instead of setters
            String publicUrl = storageService.generatePublicUrl(user.getVerificationVideoUrl());
            
            // FIX: Use the new dedicated timestamp, with a fallback for old data
            var submittedAt = user.getVerificationSubmittedAt() != null ? 
                user.getVerificationSubmittedAt() : user.getUpdatedAt();
            
            return new VerificationQueueDto(
                user.getUserId(),
                user.getFirstName(),
                user.getProfilePhotoUrl(),
                publicUrl,
                submittedAt
            );
        }).collect(Collectors.toList());
    }

    /**
     * Approves a user and sets their gender.
     */
    @Transactional
    public void approveVerification(UUID userId, VerifiedGender gender) {
        User user = userRepository.findById(userId)
                .orElseThrow(() -> new ResourceNotFoundException("User", "userId", userId));

        if (user.getVerificationStatus() != VerificationStatus.pending_review) {
            throw new IllegalStateException("User is not waiting for approval.");
        }

        // âœ… Store old video key before clearing it
        String oldVideoKey = user.getVerificationVideoUrl();

        user.setVerificationStatus(VerificationStatus.verified);
        user.setVerifiedGender(gender);
        
        // âœ… UNCOMMENTED: Delete video URL for data protection
        user.setVerificationVideoUrl(null); 

        userRepository.save(user);

        // âœ… Delete the actual file from storage using the key
        if (oldVideoKey != null) {
            try {
                storageService.deleteFile(oldVideoKey);
                log.info("Verification video deleted for user: {}", userId);
            } catch (IOException e) {
                log.error("Error deleting verification video for user {}: {}", userId, e.getMessage());
                // Don't throw exception - the user is still approved, just log the error
            }
        }

        log.info("User {} has been verified. Gender: {}", userId, gender);

        // Send push notification
        pushNotificationService.sendVerificationApproved(user);
    }

    /**
     * Rejects a user for a specific reason.
     */
    @Transactional
    public void rejectVerification(UUID userId, RejectionReason reason) {
        // Map the safe DTO enum to the persistent enum
        VerificationStatus verificationStatus;
        switch (reason) {
            case BAD_VIDEO:
                verificationStatus = VerificationStatus.rejected_bad_video;
                break;
            case MISMATCH:
                verificationStatus = VerificationStatus.rejected_mismatch;
                break;
            default:
                // This case should be impossible due to @NotNull
                throw new IllegalArgumentException("Invalid rejection reason.");
        }

        User user = userRepository.findById(userId)
                .orElseThrow(() -> new ResourceNotFoundException("User", "userId", userId));

        if (user.getVerificationStatus() != VerificationStatus.pending_review) {
            throw new IllegalStateException("User is not waiting for approval.");
        }

        // âœ… Store old video key before clearing it
        String oldVideoKey = user.getVerificationVideoUrl();

        user.setVerificationStatus(verificationStatus); // Use the mapped variable
        user.setVerificationVideoUrl(null); // Delete reference
        user.setProfileVisibility(ProfileVisibility.all); // Reset profile visibility to default

        userRepository.save(user);

        // âœ… Delete the actual file from storage using the key
        if (oldVideoKey != null) {
            try {
                storageService.deleteFile(oldVideoKey);
                log.info("Verification video deleted for rejected user: {}", userId);
            } catch (IOException e) {
                log.error("Error deleting verification video for rejected user {}: {}", userId, e.getMessage());
                // Don't throw exception - the rejection still proceeds
            }
        }

        log.info("User {} has been rejected. Reason: {}", userId, verificationStatus);

        // Send push notification about rejection
        pushNotificationService.sendVerificationRejected(user, verificationStatus);
    }

    /**
     * (For Appsmith) Bans a user (status = banned).
     */
    @Transactional
    public void banUser(UUID userId) {
        User user = userRepository.findById(userId)
                .orElseThrow(() -> new ResourceNotFoundException("User", "userId", userId));

        user.setStatus(UserStatus.banned);
        userRepository.save(user);

        log.warn("Admin has banned user {}.", userId);
    }

    /**
     * (For Appsmith) Retrieves the queue of reported users.
     */
    public List<Report> getPendingReports() {
        return reportRepository.findByStatusOrderByPriorityDesc(ReportStatus.pending);
    }
}

package com.sahbifit.api.auth;

import java.time.OffsetDateTime;
import java.util.Random;

import org.springframework.context.event.EventListener;
import org.springframework.retry.annotation.Backoff;
import org.springframework.retry.annotation.Retryable;
import org.springframework.scheduling.annotation.Async;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import com.sahbifit.api.external.sms.SmsService;
import com.sahbifit.api.user.model.User;
import com.sahbifit.api.user.repository.UserRepository;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;

@Slf4j
@Service
@RequiredArgsConstructor
public class AuthEventListener {

    private final UserRepository userRepository = null;
    private final SmsService smsService = null;

    @Async
    @Retryable(
        value = { Exception.class }, 
        maxAttempts = 3, 
        backoff = @Backoff(delay = 2000, multiplier = 2)
    )
    @EventListener
    @Transactional
    public void handleUserRegistration(UserRegisteredEvent event) {
        User user = event.getUser();
        log.info("Async: Processing registration event for user {}", user.getUserId());

        try {
            // 1. Generate OTP
            String otp = generateOtp();
            
            // 2. Save OTP to user
            // We fetch a fresh instance of the user to avoid detached entity issues
            User freshUser = userRepository.findById(user.getUserId())
                    .orElseThrow(() -> new RuntimeException("User not found for event"));

            freshUser.setSmsOtp(otp);
            freshUser.setSmsOtpExpiresAt(OffsetDateTime.now().plusMinutes(10));
            userRepository.save(freshUser);

            // 3. Send SMS
            smsService.sendVerificationSms(freshUser.getPhoneNumber(), otp);
            log.info("Async: SMS sent successfully to user {}", freshUser.getUserId());

        } catch (Exception e) {
            log.error("Async Error: Failed to send SMS for user {} after retries: {}", 
                user.getUserId(), e.getMessage(), e);
            // The @Retryable will handle re-triggering.
            // After all retries fail, it will just log the error.
            throw e; // Re-throw to trigger retry
        }
    }

    private String generateOtp() {
        // Generates a 6-digit code
        return String.format("%06d", new Random().nextInt(999999));
    }
}

@Slf4j
@Service
public class AuthService {

    @Autowired
    private AuthenticationManager authenticationManager;

    @Autowired
    private UserRepository userRepository;

    @Autowired
    private PasswordEncoder passwordEncoder;

    @Autowired
    private JwtTokenProvider tokenProvider;

    @Autowired
    private ReCaptchaService reCaptchaService;

    @Autowired
    private GymRepository gymRepository;

    @Autowired
    private TagRepository tagRepository;

    @Autowired
    private ApplicationEventPublisher eventPublisher;

    @Transactional
public void register(RegisterRequest request) {
    // 1. reCAPTCHA v3 validation
    reCaptchaService.validate(request.recaptchaToken());

    // 2. Tunisian number validation
    if (!isTunisianNumber(request.phoneNumber())) {
        throw new IllegalArgumentException("Invalid Tunisian phone number format.");
    }

    // 3. Check if user already exists
    if (userRepository.findByPhoneNumber(request.phoneNumber()).isPresent()) {
        throw new PhoneNumberAlreadyExistsException("Phone number is already registered.");
    }

    // 4. Create user object
    User user = new User();
    user.setPhoneNumber(request.phoneNumber());
    user.setPassword(passwordEncoder.encode(request.password()));
    user.setFirstName(request.firstName());
    user.setDateOfBirth(LocalDate.parse(request.dateOfBirth()));

    // 5. Set primary gym
    if (request.primaryGymId() != null) {
        Gym primaryGym = gymRepository.findById(request.primaryGymId())
                .orElseThrow(() -> new ResourceNotFoundException("Gym", "gymId", request.primaryGymId()));
        user.setPrimaryGym(primaryGym);
    }

    // 6. Process profile tags (Goals, Schedule) with OPTIMIZED single query
    java.util.Set<UserTag> allUserTags = new java.util.HashSet<>();

    // --- COMBINED TAG PROCESSING (Single DB Query) ---
    java.util.Set<Integer> allTagIds = new java.util.HashSet<>();
    
    if (request.goalTagIds() != null) {
        allTagIds.addAll(request.goalTagIds());
    }
    
    if (request.scheduleTagIds() != null) {
        allTagIds.addAll(request.scheduleTagIds());
    }

    if (!allTagIds.isEmpty()) {
        // SINGLE DATABASE QUERY for all tags
        java.util.List<Tag> allTags = tagRepository.findAllById(allTagIds);
        
        // Validation: Ensure all requested IDs were found
        if (allTags.size() != allTagIds.size()) {
            throw new ResourceNotFoundException("One or more tags were not found.");
        }

        // Create a map for fast lookup
        java.util.Map<Integer, Tag> tagMap = allTags.stream()
            .collect(Collectors.toMap(Tag::getTagId, tag -> tag));

        // --- Process GOAL tags from map ---
        if (request.goalTagIds() != null && !request.goalTagIds().isEmpty()) {
            java.util.Set<UserTag> goalUserTags = request.goalTagIds().stream()
                .map(tagId -> {
                    Tag tag = tagMap.get(tagId);
                    // Tag type validation
                    if (tag.getTagType() != com.sahbifit.api.tag.model.TagType.goal) {
                         throw new IllegalArgumentException("Tag " + tag.getTagName() + " is not a 'Goal' tag.");
                    }
                    UserTag userTag = new UserTag();
                    userTag.setUser(user);
                    userTag.setTag(tag);
                    return userTag;
                })
                .collect(Collectors.toSet());
            
            allUserTags.addAll(goalUserTags);
        }

        // --- Process SCHEDULE tags from map ---
        if (request.scheduleTagIds() != null && !request.scheduleTagIds().isEmpty()) {
            java.util.Set<UserTag> scheduleUserTags = request.scheduleTagIds().stream()
                .map(tagId -> {
                    Tag tag = tagMap.get(tagId);
                    // Tag type validation
                    if (tag.getTagType() != com.sahbifit.api.tag.model.TagType.schedule) {
                         throw new IllegalArgumentException("Tag " + tag.getTagName() + " is not a 'Schedule' tag.");
                    }
                    UserTag userTag = new UserTag();
                    userTag.setUser(user);
                    userTag.setTag(tag);
                    return userTag;
                })
                .collect(Collectors.toSet());

            allUserTags.addAll(scheduleUserTags);
        }
    }

    // Here we call the ONE correct set method from User.java
    user.setUserTags(allUserTags);

    // 7. Set default values (according to your specification)
    user.setRole(UserRole.member);
    user.setStatus(UserStatus.active);
    user.setVerificationStatus(VerificationStatus.not_started);
    user.setVerifiedGender(VerifiedGender.not_verified);
    user.setPhoneVerified(false);

    // 8. Save user
    userRepository.save(user);

    // 9. Publish event for async SMS sending
    try {
        eventPublisher.publishEvent(new UserRegisteredEvent(this, user));
        log.info("Successfully published UserRegisteredEvent for user {}", user.getPhoneNumber());
    } catch (Exception e) {
        // Log error but do not fail the registration
        log.error("Failed to publish UserRegisteredEvent for user {}: {}", user.getPhoneNumber(), e.getMessage());
    }
}

    public AuthResponse login(LoginRequest loginRequest) {
        // REMOVED: The manual user check and phone verification check
        // All validation will now happen in UserDetailsServiceImpl
        
        // 1. Authenticate (this will call UserDetailsServiceImpl.loadUserByUsername)
        Authentication authentication = authenticationManager.authenticate(
                new UsernamePasswordAuthenticationToken(
                        loginRequest.phoneNumber(),
                        loginRequest.password()
                )
        );

        // 2. Set Security Context
        SecurityContextHolder.getContext().setAuthentication(authentication);

        // 3. Generate JWT
        String jwt = tokenProvider.generateToken(authentication);
        return new AuthResponse(jwt);
    }

    /**
     * Verifies a user's SMS OTP code.
     */
    @Transactional
    public void verifySmsOtp(VerifySmsRequest request) {
        // 1. Find user by phone number
        User user = userRepository.findByPhoneNumber(request.phoneNumber())
                .orElseThrow(() -> new ResourceNotFoundException("User", "phoneNumber", request.phoneNumber()));

        // 2. Check if already verified
        if (user.isPhoneVerified()) {
            // No error, but no action needed.
            return; 
        }

        // 3. Check if OTP is stored
        if (user.getSmsOtp() == null || user.getSmsOtpExpiresAt() == null) {
            throw new InvalidOtpException("No pending OTP found. Please request a new one.");
        }

        // 4. Check if OTP has expired
        if (OffsetDateTime.now().isAfter(user.getSmsOtpExpiresAt())) {
            // Delete OTP to prevent reuse
            user.setSmsOtp(null);
            user.setSmsOtpExpiresAt(null);
            userRepository.save(user);
            throw new OtpExpiredException("OTP has expired. Please request a new code.");
        }

        // 5. Check if OTP code is correct
        if (!user.getSmsOtp().equals(request.otp())) {
            throw new InvalidOtpException("Invalid OTP code.");
        }

        // 6. Success! Mark user as verified
        user.setPhoneVerified(true);
        
        // 7. Clear OTP fields for security
        user.setSmsOtp(null);
        user.setSmsOtpExpiresAt(null);

        userRepository.save(user);
    }

    /**
     * Sends a new OTP code to an unverified phone number.
     * Used when the original OTP has expired.
     */
    @Transactional
    public void resendOtp(ResendOtpRequest request) {
        // 1. Validate reCAPTCHA
        reCaptchaService.validate(request.recaptchaToken());

        // 2. Tunisian number validation
        if (!isTunisianNumber(request.phoneNumber())) {
            throw new IllegalArgumentException("Invalid Tunisian phone number format.");
        }

        // 3. Find user by phone number
        User user = userRepository.findByPhoneNumber(request.phoneNumber())
                .orElseThrow(() -> new ResourceNotFoundException("User", "phoneNumber", request.phoneNumber()));

        // 4. Check if already verified
        if (user.isPhoneVerified()) {
            throw new IllegalArgumentException("Account is already verified. Please log in.");
        }

        // 5. Generate new OTP (this will be handled by AuthEventListener)
        // Note: OTP generation is now handled asynchronously by AuthEventListener

            return documentId;
        } catch (InterruptedException | ExecutionException e) {

        }
    }

package com.sahbifit.api.external.recaptcha;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import org.springframework.util.LinkedMultiValueMap;
import org.springframework.util.MultiValueMap;
import org.springframework.web.client.RestTemplate;

import com.sahbifit.api.config.ReCaptchaConfig;

@Service
public class GoogleReCaptchaService implements ReCaptchaService {

    private static final Logger logger = LoggerFactory.getLogger(GoogleReCaptchaService.class);

    private final ReCaptchaConfig reCaptchaConfig;
    private final RestTemplate restTemplate;

    @Autowired
    public GoogleReCaptchaService(ReCaptchaConfig reCaptchaConfig, RestTemplate restTemplate) {
        this.reCaptchaConfig = reCaptchaConfig;
        // We are reusing the RestTemplate bean created in WinSmsConfig!
        this.restTemplate = restTemplate; 
    }

    @Override
    public void validate(String token) {
        // Google expects a POST request with 'x-www-form-urlencoded' data
        MultiValueMap<String, String> body = new LinkedMultiValueMap<>();
        body.add("secret", reCaptchaConfig.getSecretKey());
        body.add("response", token);
        // Optionally, you can also send the user's IP:
        // body.add("remoteip", userIp);

        ReCaptchaResponse response;
        try {
            response = restTemplate.postForObject(
                    reCaptchaConfig.getVerifyUrl(),
                    body,
                    ReCaptchaResponse.class
            );
        } catch (Exception e) {
            logger.error("Error communicating with Google reCAPTCHA API: {}", e.getMessage());
            // We throw an exception to stop the registration
            throw new RuntimeException("reCAPTCHA service unavailable.");
        }

        // 1. Check if the request itself was successful
        if (response == null || !response.isSuccess()) {
            logger.warn("reCAPTCHA validation failed. Errors: {}", 
                (response != null ? response.getErrorCodes() : "No response"));
            throw new RuntimeException("reCAPTCHA validation failed.");
        }

        // 2. Check if the score (bot probability) is sufficient
        if (response.getScore() < reCaptchaConfig.getScoreThreshold()) {
            logger.warn("Potential bot blocked. reCAPTCHA Score: {}", response.getScore());
            throw new RuntimeException("Validation failed (low score).");
        }
        
        // 3. (Optional) You can also check the 'action' 
        //    to ensure the token comes from your registration page.
        //    e.g., if (!"register".equals(response.getAction())) { ... }

        logger.info("reCAPTCHA validation successful (Score: {}).", response.getScore());
    }
}

    private static final Logger logger = LoggerFactory.getLogger(WinSmsService.class);

    private final WinSmsConfig winSmsConfig;
    private final RestTemplate restTemplate;

    @Autowired
    public WinSmsService(WinSmsConfig winSmsConfig, RestTemplate restTemplate) {
        this.winSmsConfig = winSmsConfig;
        this.restTemplate = restTemplate;
    }

    @Override
    public void sendVerificationSms(String phoneNumber, String otp) {
        
        // ðŸš¨ IMPORTANT: WinSMS expects '216' as prefix.
        // Your validation stores the number without prefix.
        String fullPhoneNumber = "216" + phoneNumber;
        
        String message = String.format("Your SahbiFit verification code is: %s", otp);

        try {
            // We build the URL exactly like in your example,
            // since WinSMS might expect special URL encoding.
            String params = String.format("action=send-sms&api_key=%s&to=%s&sms=%s",
                    URLEncoder.encode(winSmsConfig.getApiKey(), StandardCharsets.UTF_8),
                    URLEncoder.encode(fullPhoneNumber, StandardCharsets.UTF_8),
                    URLEncoder.encode(message, StandardCharsets.UTF_8));
            
            String fullUrl = winSmsConfig.getBaseUrl() + "?" + params;

            if (response != null && response.contains("OK")) {
                logger.info("SMS to {} sent successfully. (OTP: {})", fullPhoneNumber, otp);
            } else {
                // If "OK" is not in the response, log the error
                logger.warn("Error sending SMS to {}. API response: {}", fullPhoneNumber, response);
            }

        } catch (Exception e) {
            // Catches all errors (e.g., java.io.UnsupportedEncodingException or RestClientException)
            logger.error("Error connecting to WinSMS API: {}", e.getMessage(), e);
        }
    }

package com.sahbifit.api.external.storage;

import lombok.extern.slf4j.Slf4j;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import org.springframework.web.multipart.MultipartFile;

import software.amazon.awssdk.core.sync.RequestBody;
import software.amazon.awssdk.services.s3.S3Client;
import software.amazon.awssdk.services.s3.model.DeleteObjectRequest;
import software.amazon.awssdk.services.s3.model.PutObjectRequest;
import software.amazon.awssdk.services.s3.model.S3Exception;

import java.io.IOException;
import java.util.UUID;

import com.sahbifit.api.config.CloudflareConfig;

@Slf4j
@Service
public class CloudflareR2Service implements StorageService {

    @Autowired
    private S3Client s3Client;
    
    @Autowired
    private CloudflareConfig cloudflareConfig;

    @Override
    public String uploadFile(MultipartFile file, String folder, String fileName) throws IOException {
        try {
            // File validation
            if (file.isEmpty()) {
                throw new IllegalArgumentException("File is empty");
            }

            // Generate the complete S3 key (path in the bucket)
            String key = generateFileKey(folder, fileName);
            
            log.info("Uploading file: {} -> {}/{}", file.getOriginalFilename(), 
                    cloudflareConfig.getBucketName(), key);

            // Metadata for the video
            var metadata = new java.util.HashMap<String, String>();
            metadata.put("Content-Type", file.getContentType());
            metadata.put("Original-Filename", file.getOriginalFilename());
            metadata.put("Upload-Timestamp", String.valueOf(System.currentTimeMillis()));

            // Create PutObjectRequest
            PutObjectRequest putObjectRequest = PutObjectRequest.builder()
                    .bucket(cloudflareConfig.getBucketName())
                    .key(key)
                    .contentType(file.getContentType())
                    .contentLength(file.getSize())
                    .metadata(metadata)
                    .build();

            // Upload file
            s3Client.putObject(putObjectRequest, 
                    RequestBody.fromBytes(file.getBytes()));
            
            log.info("File successfully uploaded: {}/{}", cloudflareConfig.getBucketName(), key);
            return key; // RETURN THE KEY INSTEAD OF PUBLIC URL

        } catch (S3Exception e) {
            log.error("S3 error uploading file: {}", e.awsErrorDetails().errorMessage(), e);
            throw new IOException("Error uploading file: " + e.awsErrorDetails().errorMessage(), e);
        } catch (Exception e) {
            log.error("Unexpected error uploading file", e);
            throw new IOException("Unexpected error during upload: " + e.getMessage(), e);
        }
    }

    @Override
    public void deleteFile(String fileKey) throws IOException {
        try {
            log.info("Deleting file: {}", fileKey);
            
            DeleteObjectRequest deleteObjectRequest = DeleteObjectRequest.builder()
                    .bucket(cloudflareConfig.getBucketName())
                    .key(fileKey) // USE THE KEY DIRECTLY
                    .build();

            s3Client.deleteObject(deleteObjectRequest);
            
            log.info("File successfully deleted: {}", fileKey);

        } catch (S3Exception e) {
            log.error("S3 error deleting file: {}", e.awsErrorDetails().errorMessage(), e);
            throw new IOException("Error deleting file: " + e.awsErrorDetails().errorMessage(), e);
        }
    }

    /**
     * Generates the public URL for the file
     * Now public so AdminService can use it
     */
    @Override
    public String generatePublicUrl(String key) {
        if (key == null) return null;
        return String.format("%s/%s", cloudflareConfig.getPublicUrl(), key);
    }

    /**
     * Generates the S3 key for the file
     */
    private String generateFileKey(String folder, String fileName) {
        // Remove leading and trailing slashes
        String cleanFolder = folder.replaceAll("^/|/$", "");
        return String.format("%s/%s", cleanFolder, fileName);
    }

    /**
     * Helper method to generate a unique filename
     */
    public String generateUniqueFileName(String originalFileName) {
        String extension = "";
        if (originalFileName != null && originalFileName.contains(".")) {
            extension = originalFileName.substring(originalFileName.lastIndexOf("."));
        }
        return UUID.randomUUID().toString() + extension;
    }
}

    @Transactional(readOnly = true)

    }

     * (Method 2) For Users: Submit a new gym for admin review.
     * (MVP 7.5)

                .orElseThrow(() -> new ResourceNotFoundException("User not found"));

        Gym gym = new Gym();
        gym.setName(dto.name());
        gym.setAddress(dto.address());
        // Note: Location (Point) is not set here per the plan
        gym.setStatus(GymStatus.pending); // Set to pending for review
        gym.setSubmittedByUser(user);

        return gymRepository.save(gym);
    }

        gym.setName(dto.name());
        gym.setAddress(dto.address());
        gym.setStatus(GymStatus.approved); // Approved immediately
        // submittedByUser is null for admin-created gyms

        return gymRepository.save(gym);
    }

     * (Method 4) For Admins: Approve a user-submitted gym.
     * (MVP 7.5)

                .orElseThrow(() -> new ResourceNotFoundException("Gym not found with ID: " + gymId));

        return gymRepository.save(gym);

                .orElseThrow(() -> new ResourceNotFoundException("Gym not found with ID: " + gymId));

     */
    @Transactional(readOnly = true)
    public List<Gym> getPendingGyms() {
        return gymRepository.findByStatus(GymStatus.pending);
    }
}

package com.sahbifit.api.rating;

import com.sahbifit.api.external.push.PushNotificationService;
import com.sahbifit.api.session.model.ParticipantStatus;
import com.sahbifit.api.session.model.Session;
import com.sahbifit.api.session.model.SessionParticipant;
import com.sahbifit.api.session.repository.SessionParticipantRepository;
import com.sahbifit.api.session.repository.SessionRepository;
import com.sahbifit.api.user.model.User;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.OffsetDateTime;
import java.util.List;
import java.util.Set;
import java.util.stream.Collectors;

@Slf4j
@Service
@RequiredArgsConstructor
public class RatingNotifierService {

    private final SessionRepository sessionRepository;
    private final SessionParticipantRepository sessionParticipantRepository;
    private final PushNotificationService pushNotificationService;

    /**
     * Runs every 15 minutes to check for completed sessions
     * and trigger rating push notifications.
     */
    @Scheduled(fixedRate = 900000) // 900000ms = 15 minutes
    @Transactional
    public void triggerPendingRatings() {
        log.info("Running pending rating notifier job...");

        // 1. Find sessions that ended and haven't been processed
        List<Session> sessionsToNotify = sessionRepository
                .findCompletedSessionsPendingNotification(OffsetDateTime.now());

        if (sessionsToNotify.isEmpty()) {
            log.info("No sessions found for rating notification.");
            return;
        }

        log.info("Found {} sessions to process for rating notifications.", sessionsToNotify.size());

        for (Session session : sessionsToNotify) {
            try {
                // 2. Find all users who were marked as 'attended'
                List<SessionParticipant> attendees = sessionParticipantRepository
                        .findBySessionAndStatus(session, ParticipantStatus.attended);

                // No need to send notifications if fewer than 2 people attended
                if (attendees.size() < 2) {
                    log.warn("Skipping session {} (not enough attendees)", session.getSessionId());
                } else {
                    // 3. Send a push notification to each attendee
                    Set<User> attendeeUsers = attendees.stream()
                            .map(SessionParticipant::getUser)
                            .collect(Collectors.toSet());
                    
                    for (User user : attendeeUsers) {
                        pushNotificationService.sendRatingReminder(user, session);
                    }
                }
                
                // 4. Mark the session as processed so we don't send notifications again
                session.setRatingNotified(true);
                sessionRepository.save(session);

            } catch (Exception e) {
                log.error("Failed to process rating notification for session {}: {}",
                        session.getSessionId(), e.getMessage(), e);
                // Continue to the next session
            }
        }

        log.info("Rating notifier job finished.");
    }

import java.util.List;

import org.springframework.stereotype.Service;

import com.sahbifit.api.session.repository.SessionRepository;
import com.sahbifit.api.user.model.User;
import com.sahbifit.api.user.repository.UserRepository;

import lombok.RequiredArgsConstructor;

@Service
@RequiredArgsConstructor
@Transactional
public class RatingService {

    private final RatingRepository ratingRepository;
    private final UserRepository userRepository;
    private final SessionRepository sessionRepository; // Required to find the session
    private final SessionParticipantRepository sessionParticipantRepository;

    /**
     * (MVP 5.1) Creates a new rating of one user for another
     * after a shared session.
     */
    public Rating createRating(CreateRatingDto dto, String raterPhoneNumber) {
        // 1. Find user and session
        User raterUser = userRepository.findByPhoneNumber(raterPhoneNumber)
                .orElseThrow(() -> new ResourceNotFoundException("Rater (User) not found"));

        User ratedUser = userRepository.findById(dto.ratedUserId())
                .orElseThrow(() -> new ResourceNotFoundException("Rated User not found"));

        Session session = sessionRepository.findById(dto.sessionId())
                .orElseThrow(() -> new ResourceNotFoundException("Session not found"));

        // Validation 0 (Rating yourself)
        if (raterUser.getUserId().equals(ratedUser.getUserId())) {
            throw new IllegalStateException("You cannot rate yourself.");
        }

        // Validation 1 (Participation)
        SessionParticipantId raterId = new SessionParticipantId();
        raterId.setSessionId(dto.sessionId());
        raterId.setUserId(raterUser.getUserId());

        SessionParticipantId ratedId = new SessionParticipantId();
        ratedId.setSessionId(dto.sessionId());
        ratedId.setUserId(dto.ratedUserId());

        if (!sessionParticipantRepository.existsById(raterId) || !sessionParticipantRepository.existsById(ratedId)) {
            throw new IllegalStateException("Both users must have participated in this session to rate each other.");
        }

        // Validation 2 (Duplicate rating)
        boolean alreadyExists = ratingRepository.existsBySessionSessionIdAndRatingByUserIdAndRatingForUserId(
                dto.sessionId(),
                raterUser.getUserId(),
                dto.ratedUserId()
        );

        if (alreadyExists) {
            throw new IllegalStateException("You have already rated this user for this session.");
        }

        // 3. Create and save rating
        Rating newRating = new Rating();
        newRating.setSession(session);
        newRating.setRatingBy(raterUser);
        newRating.setRatingFor(ratedUser);
        newRating.setRatingValue(dto.ratingValue());
        newRating.setComment(dto.comment()); // Can be null

        return ratingRepository.save(newRating);
    }

    /**
     * (Optional) Retrieves all received ratings for a user.
     */
    @Transactional(readOnly = true)
    public List<Rating> getRatingsForUser(UUID userId) {
        // Ensure user exists before retrieving ratings
        if (!userRepository.existsById(userId)) {
            throw new ResourceNotFoundException("User not found");
        }
        return ratingRepository.findByRatingForUserId(userId);
    }

    /**
     * (MVP 5.2) Gets a list of users/sessions that the current user
     * has attended but not yet rated - OPTIMIZED VERSION
     */
    @Transactional(readOnly = true)
    public List<PendingRatingDto> getPendingRatings(String raterPhoneNumber) {
        // Validate user exists
        if (!userRepository.existsByPhoneNumber(raterPhoneNumber)) {
            throw new ResourceNotFoundException("Rater (User) not found");
        }
        
        // Single query to get all pending ratings
        return ratingRepository.findPendingRatings(raterPhoneNumber);
    }
}

    private final UserRepository userRepository;
    private final SessionRepository sessionRepository; // Required for optional sessionId

    // Defines the critical keywords
    private static final Set<String> CRITICAL_KEYWORDS = Set.of(
            "safety", "threat", "harassment", "gefahr", "bedrohung", "unsafe", 
            "belÃ¤stigung", "gewalt", "violence", "stalking"
    );

    @Transactional
    public void createReport(CreateReportDto dto, String reporterPhoneNumber) {
        // 1. Find reporter
        User reporterUser = userRepository.findByPhoneNumber(reporterPhoneNumber)
                .orElseThrow(() -> new ResourceNotFoundException("User", "phoneNumber", reporterPhoneNumber));

        // 2. Find reported user
        User reportedUser = userRepository.findById(dto.reportedUserId())
                .orElseThrow(() -> new ResourceNotFoundException("User", "userId", dto.reportedUserId()));

        // 3. Find optional session
        Session session = null;
        if (dto.sessionId() != null) {
            // We don't fail if the session ID is invalid, we just log it.
            session = sessionRepository.findById(dto.sessionId()).orElse(null);
            if (session == null) {
                log.warn("Report created with invalid sessionId: {}", dto.sessionId());
            }
        }

        // 4. Determine priority (MVP 7.2)
        ReportPriority priority = determinePriority(dto.reason(), reporterUser);

        // 5. Create report object
        Report report = new Report();
        report.setReportedBy(reporterUser);
        report.setReportedUser(reportedUser);
        report.setSession(session);
        report.setReason(dto.reason());
        report.setPriority(priority);
        report.setStatus(ReportStatus.pending); // Default status

        // 6. Save report
        reportRepository.save(report);

        // 7. (Optional) Immediate notification for critical priority
        if (priority == ReportPriority.critical) {
            log.warn("[SLEEP-SAFE TRIAGE] Critical report (ID: {}) from {} against {}. Reason: {}",
                    report.getReportId(), reporterUser.getUserId(), reportedUser.getUserId(), dto.reason());
            // Trigger event here, e.g., via ApplicationEventPublisher
            // (e.g., publishEvent(new CriticalReportEvent(this, report));)
        }
    }

    /**
     * Determines the priority of a report based on keywords and the reporter's role.
     * (MVP 7.2 logic)
     */
    private ReportPriority determinePriority(String reason, User reporter) {
        String lowerCaseReason = reason.toLowerCase();

        // 1. Check for critical keywords
        boolean isCritical = CRITICAL_KEYWORDS.stream().anyMatch(lowerCaseReason::contains);
        if (isCritical) {
            return ReportPriority.critical;
        }

        // 2. Check reporter's role
        if (reporter.getRole() == UserRole.ambassador || reporter.getRole() == UserRole.admin) {
            return ReportPriority.high;
        }

        // 3. Otherwise
        return ReportPriority.low;
    }

package com.sahbifit.api.security;

import java.util.Date;
import java.util.List;
import java.util.stream.Collectors;

import javax.crypto.SecretKey;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.security.core.Authentication;
import org.springframework.security.core.GrantedAuthority;
import org.springframework.security.core.userdetails.UserDetails;
import org.springframework.stereotype.Component;

import io.jsonwebtoken.Claims;
import io.jsonwebtoken.ExpiredJwtException;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.MalformedJwtException;
// import io.jsonwebtoken.SignatureAlgorithm; // Removed: Deprecated
import io.jsonwebtoken.UnsupportedJwtException;
import io.jsonwebtoken.security.Keys;
import io.jsonwebtoken.security.SignatureException; // Added: New exception class

@Component
public class JwtTokenProvider {

    private static final Logger logger = LoggerFactory.getLogger(JwtTokenProvider.class);

    @Value("${app.jwtSecret}")
    private String jwtSecret;

    @Value("${app.jwtExpirationInMs}")
    private int jwtExpirationInMs;

    private SecretKey getSigningKey() {
        return Keys.hmacShaKeyFor(jwtSecret.getBytes());
    }

    // Creates a JWT for an authenticated user WITH ROLES
    public String generateToken(Authentication authentication) {
        UserDetails userPrincipal = (UserDetails) authentication.getPrincipal();

        // Extract roles from Authorities
        List<String> roles = userPrincipal.getAuthorities().stream()
                .map(GrantedAuthority::getAuthority)
                .collect(Collectors.toList());

        Date now = new Date();
        Date expiryDate = new Date(now.getTime() + jwtExpirationInMs);

        return Jwts.builder()
                .subject(userPrincipal.getUsername()) // Replaced setSubject
                .claim("roles", roles) // ðŸ†• Store roles in token
                .issuedAt(new Date()) // Replaced setIssuedAt
                .expiration(expiryDate) // Replaced setExpiration
                .signWith(getSigningKey(), Jwts.SIG.HS512) // Replaced SignatureAlgorithm.HS512
                .compact();
    }

    // Extracts the phone number from the JWT
    public String getUsernameFromJWT(String token) {
        Claims claims = Jwts.parser() // Replaced parserBuilder()
                .verifyWith(getSigningKey()) // Replaced setSigningKey()
                .build()
                .parseSignedClaims(token) // Replaced parseClaimsJws()
                .getPayload(); // Replaced getBody()

        return claims.getSubject();
    }

    // ðŸ†• Extracts roles from JWT token
    public List<String> getRolesFromJWT(String token) {
        Claims claims = Jwts.parser() // Replaced parserBuilder()
                .verifyWith(getSigningKey()) // Replaced setSigningKey()
                .build()
                .parseSignedClaims(token) // Replaced parseClaimsJws()
                .getPayload(); // Replaced getBody()

        return claims.get("roles", List.class);
    }

    // Validates a JWT
    public boolean validateToken(String authToken) {
        try {
            Jwts.parser() // Replaced parserBuilder()
                .verifyWith(getSigningKey()) // Replaced setSigningKey()
                .build()
                .parseSignedClaims(authToken); // Replaced parseClaimsJws()
            return true;
        } catch (SignatureException ex) { // Using new security.SignatureException
            logger.error("Invalid JWT signature");
        } catch (MalformedJwtException ex) {
            logger.error("Invalid JWT token");
        } catch (ExpiredJwtException ex) {
            logger.error("Expired JWT token");
        } catch (UnsupportedJwtException ex) {
            logger.error("Unsupported JWT token");
        } catch (IllegalArgumentException ex) {
            logger.error("JWT claims string is empty.");
        }
        return false;
    }
}

package com.sahbifit.api.session;

import com.sahbifit.api.session.model.Session;
import com.sahbifit.api.session.repository.SessionRepository;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.OffsetDateTime;
import java.util.List;

@Slf4j
@Service
@RequiredArgsConstructor
public class ChatActivationService {

    private final SessionRepository sessionRepository;

    // Activation window: 1 hour (60 minutes)
    private static final int ACTIVATION_WINDOW_MINUTES = 60;
    
    // Deactivation window: 1 hour (60 minutes)
    private static final int DEACTIVATION_WINDOW_MINUTES = 60;

    // Scheduler runs every 10 minutes (600,000 milliseconds)
    private static final long JOB_RATE_MS = 600000;

    /**
     * Runs every 10 minutes to activate chats for upcoming sessions.
     */
    @Scheduled(fixedRate = JOB_RATE_MS) 
    @Transactional
    public void activateUpcomingChats() {
        OffsetDateTime now = OffsetDateTime.now();
        OffsetDateTime activationTime = now.plusMinutes(ACTIVATION_WINDOW_MINUTES);

        log.info("[ChatScheduler] Running activation job. Finding sessions starting before: {}", activationTime);

        List<Session> sessionsToActivate = sessionRepository.findSessionsToActivate(now, activationTime);

        if (sessionsToActivate.isEmpty()) {
            log.info("[ChatScheduler] No sessions found for chat activation.");
            return;
        }

        log.info("[ChatScheduler] Activating chat for {} sessions.", sessionsToActivate.size());
        for (Session session : sessionsToActivate) {
            session.setChatActive(true);
            // We don't set isChatProcessed yet; that's for the deactivation job
        }
        
        // Batch save
        sessionRepository.saveAll(sessionsToActivate);
    }

    /**
     * Runs every 10 minutes to deactivate chats for completed sessions.
     */
    @Scheduled(fixedRate = JOB_RATE_MS) 
    @Transactional
    public void deactivateCompletedChats() {
        OffsetDateTime now = OffsetDateTime.now();
        OffsetDateTime deactivationTime = now.minusMinutes(DEACTIVATION_WINDOW_MINUTES);

        log.info("[ChatScheduler] Running deactivation job. Finding sessions ending before: {}", deactivationTime);

        List<Session> sessionsToDeactivate = sessionRepository.findSessionsToDeactivate(deactivationTime);

        if (sessionsToDeactivate.isEmpty()) {
            log.info("[ChatScheduler] No sessions found for chat deactivation.");
            return;
        }

        log.info("[ChatScheduler] Deactivating chat for {} sessions.", sessionsToDeactivate.size());
        for (Session session : sessionsToDeactivate) {
            session.setChatActive(false);
            session.setChatProcessed(true); // Mark as done to prevent re-processing
        }
        
        // Batch save
        sessionRepository.saveAll(sessionsToDeactivate);
    }
}

@Slf4j
@Service
@RequiredArgsConstructor
@Transactional
public class SessionService {

    private final SessionRepository sessionRepository;
    private final UserRepository userRepository;
    private final GymRepository gymRepository;
    private final SessionParticipantRepository sessionParticipantRepository;
    private final UserService userService;
    private final ApplicationEventPublisher eventPublisher;

    /**
     * (Method 1) Create a new session, applying all "Create" gates.
     */
    public Session createSession(CreateSessionDto dto, String userPhoneNumber) {
        User user = userRepository.findByPhoneNumber(userPhoneNumber)
                .orElseThrow(() -> new ResourceNotFoundException("User", "phoneNumber", userPhoneNumber));

        // Gate 1 (Verification)
        if (user.getVerificationStatus() != VerificationStatus.verified) {
            throw new IllegalStateException("You must be fully verified to create a session.");
        }

        // Gate 2 (Reputation - "The 1-Session Gate" Strategy)
        if (dto.sessionType() == SessionType.one_on_one && user.getGroupSessionCount() < 1) { // <-- CHANGED FROM 3 to 1
            throw new IllegalStateException("You must attend 1 group session before creating a 1-on-1 session."); // <-- UPDATED MESSAGE
        }

        Gym gym = gymRepository.findById(dto.gymId())
                .orElseThrow(() -> new ResourceNotFoundException("Gym", "gymId", dto.gymId()));

        // Create the session (don't save yet)
        Session session = new Session();
        session.setCreator(user);
        session.setGym(gym);
        session.setSessionType(dto.sessionType());
        session.setStatus(SessionStatus.active);
        session.setTitle(dto.title());
        session.setStartTime(dto.startTime());
        session.setEndTime(dto.endTime());
        session.setMaxParticipants(dto.maxParticipants());

        // Save the session (This is where the session.sessionId is generated)
        Session savedSession = sessionRepository.save(session);

        // Add the creator as the first participant
        SessionParticipantId creatorId = new SessionParticipantId();
        creatorId.setSessionId(savedSession.getSessionId());
        creatorId.setUserId(user.getUserId());

        SessionParticipant creatorEntry = new SessionParticipant();
        creatorEntry.setId(creatorId);
        creatorEntry.setSession(savedSession);
        creatorEntry.setUser(user);
        creatorEntry.setStatus(ParticipantStatus.joined);
        sessionParticipantRepository.save(creatorEntry);

        // PUBLISH THE EVENT for the async listener
        try {
            // We pass 'savedSession', which now has a non-null ID
            eventPublisher.publishEvent(new SessionCreatedEvent(this, savedSession));
        } catch (Exception e) {
            log.error("Failed to publish SessionCreatedEvent for session {}: {}", savedSession.getSessionId(), e.getMessage());
            // Log error, but don't fail the transaction
        }

        return savedSession;
    }

    /**
     * (Method 2) Join an existing session, applying all "Join" gates.
     */
    public void joinSession(UUID sessionId, String userPhoneNumber) {
        User user = userRepository.findByPhoneNumber(userPhoneNumber)
                .orElseThrow(() -> new ResourceNotFoundException("User", "phoneNumber", userPhoneNumber));
        
        Session session = sessionRepository.findById(sessionId)
                .orElseThrow(() -> new ResourceNotFoundException("Session", "sessionId", sessionId));

        // Gate 1 (Probationary Mode - MVP 2.6)
        if (user.getVerificationStatus() == VerificationStatus.pending_review) {
            if (session.getCreator().getRole() != UserRole.ambassador) {
                throw new IllegalStateException("Verification pending. You can only join Ambassador sessions.");
            }
        }

        // Gate 2 (Reputation - "The 1-Session Gate" Strategy)
        if (session.getSessionType() == SessionType.one_on_one && user.getGroupSessionCount() < 1) { // <-- CHANGED FROM 3 to 1
            throw new IllegalStateException("You must attend 1 group session before joining a 1-on-1 session."); // <-- UPDATED MESSAGE
        }

        // Gate 3 (Session Full?)
        int participantCount = sessionParticipantRepository.countBySessionId(sessionId);
        if (participantCount >= session.getMaxParticipants()) {
            throw new IllegalStateException("This session is already full.");
        }

        // Gate 4 (Already Joined?)
        SessionParticipantId participantId = new SessionParticipantId();
        participantId.setSessionId(sessionId);
        participantId.setUserId(user.getUserId());

        if (sessionParticipantRepository.existsById(participantId)) {
            throw new IllegalStateException("You have already joined this session.");
        }

        // Success: Create and save the participant entry
        SessionParticipant newParticipant = new SessionParticipant();
        newParticipant.setId(participantId);
        newParticipant.setSession(session);
        newParticipant.setUser(user);
        newParticipant.setStatus(ParticipantStatus.joined);
        sessionParticipantRepository.save(newParticipant);
    }

    /**
     * (Method 3) Get all active sessions for the feed.
     * (MVP 6.1)
     * NOW INCLUDES GENDER FILTERING AND PRIVACY/VISIBILITY LOGIC.
     * OPTIMIZED: Uses JOIN FETCH to prevent N+1 queries
     */
    @Transactional(readOnly = true)
    public List<Session> getActiveSessions(String userPhoneNumber, VerifiedGender genderFilter) {
        
        // 1. Get the user who is VIEWING the feed
        User viewer = userRepository.findByPhoneNumber(userPhoneNumber)
                .orElseThrow(() -> new ResourceNotFoundException("User", "phoneNumber", userPhoneNumber));

        List<Session> sessions;

        // 2. Apply the GENDER FILTER (existing logic) using optimized queries
        if (genderFilter != null) {
            // Security Check: You can only filter by your own gender.
            if (viewer.getVerifiedGender() != genderFilter) {
                throw new AccessDeniedException("You can only filter sessions by your own verified gender.");
            }
            // Use optimized query with JOIN FETCH
            sessions = sessionRepository.findByStatusAndCreatorVerifiedGenderWithCreatorAndGym(SessionStatus.active, genderFilter);
        } else {
            // Use optimized query with JOIN FETCH
            sessions = sessionRepository.findByStatusWithCreatorAndGym(SessionStatus.active);
        }

        // 3. Apply PRIVACY FILTER (new logic)
        return sessions.stream()
                .filter(session -> isVisible(session.getCreator(), viewer))
                .collect(Collectors.toList());
    }

    /**
     * Helper method to check if a creator's profile is visible to a viewer.
     */
    private boolean isVisible(User creator, User viewer) {
        // If creator is visible to all, return true.
        if (creator.getProfileVisibility() == ProfileVisibility.all) {
            return true;
        }

        // If creator is 'same_gender' only, check for match.
        if (creator.getProfileVisibility() == ProfileVisibility.same_gender) {
            // Both must be verified AND their genders must match.
            return creator.getVerifiedGender() != VerifiedGender.not_verified &&
                   creator.getVerifiedGender() == viewer.getVerifiedGender();
        }

        return false;
    }

    @Transactional
    public void markAttendance(MarkAttendanceDto dto, String markerPhoneNumber) {
        
        // 1. Find user who is marking attendance
        User markerUser = userRepository.findByPhoneNumber(markerPhoneNumber)
                .orElseThrow(() -> new ResourceNotFoundException("User", "phoneNumber", markerPhoneNumber));

        // 2. Find session
        Session session = sessionRepository.findById(dto.sessionId())
                .orElseThrow(() -> new ResourceNotFoundException("Session", "sessionId", dto.sessionId()));

        // 3. Permission check
        boolean isCreator = session.getCreator().getUserId().equals(markerUser.getUserId());
        boolean isAdmin = markerUser.getRole() == UserRole.admin;

        if (!isCreator && !isAdmin) {
            throw new AccessDeniedException("Only the session creator or an admin can mark attendance.");
        }

        // 4. Find SessionParticipant (the target object)
        SessionParticipantId participantId = new SessionParticipantId();
        participantId.setSessionId(dto.sessionId());
        participantId.setUserId(dto.participantUserId());

        SessionParticipant participant = sessionParticipantRepository.findById(participantId)
                .orElseThrow(() -> new ResourceNotFoundException("SessionParticipant", "Id", participantId));

        // 5. Update status
        participant.setStatus(dto.newStatus());
        sessionParticipantRepository.save(participant);

        // 6. Trigger chain (MVP 5.4)
        if (dto.newStatus() == ParticipantStatus.no_show) {
            // Calls logic in UserService
            userService.incrementNoShowCount(dto.participantUserId());
        } 
        else if (dto.newStatus() == ParticipantStatus.attended) {
            // Check if it was a group session (MVP 3.5)
            if (session.getSessionType() == SessionType.group) {
                userService.incrementGroupSessionCount(dto.participantUserId());
            }
        }
    }
}

package com.sahbifit.api.user;

import java.time.OffsetDateTime;
import java.util.UUID;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import com.sahbifit.api.badge.model.Badge;
import com.sahbifit.api.badge.model.UserBadge;
import com.sahbifit.api.badge.model.UserBadgeId;
import com.sahbifit.api.badge.repository.BadgeRepository;
import com.sahbifit.api.badge.repository.UserBadgeRepository;
import com.sahbifit.api.exception.ResourceNotFoundException;
import com.sahbifit.api.user.dto.UpdateFcmTokenDto;
import com.sahbifit.api.user.dto.UpdateVisibilityDto;
import com.sahbifit.api.user.model.ProfileVisibility;
import com.sahbifit.api.user.model.User;
import com.sahbifit.api.user.model.UserBlock;
import com.sahbifit.api.user.model.UserBlockId;
import com.sahbifit.api.user.model.UserStatus;
import com.sahbifit.api.user.repository.UserBlockRepository;
import com.sahbifit.api.user.repository.UserRepository;
import com.sahbifit.api.verification.model.VerifiedGender;

import lombok.RequiredArgsConstructor;

@Service
@RequiredArgsConstructor
public class UserService {

    private static final Logger log = LoggerFactory.getLogger(UserService.class);
    private final UserRepository userRepository;
    private final UserBlockRepository userBlockRepository;
    private final BadgeRepository badgeRepository;
    private final UserBadgeRepository userBadgeRepository;

    // Defines the limit for "No-Shows"
    private static final int NO_SHOW_LIMIT = 3;
    private static final String UNRELIABLE_BADGE_NAME = "UNRELIABLE";

    /**
     * Increases the counter for completed group sessions of a user.
     * (MVP 3.5 Logic)
     */
    @Transactional
    public void incrementGroupSessionCount(UUID userId) {
        User user = userRepository.findById(userId)
                .orElseThrow(() -> new ResourceNotFoundException("User", "userId", userId));
        
        int newGroupCount = user.getGroupSessionCount() + 1;
        user.setGroupSessionCount(newGroupCount);
        userRepository.save(user);
        
        log.info("User {} group session count incremented to {}.", userId, newGroupCount);
    }

    /**
     * Increases the "No-Show" counter for a user and applies a block if necessary.
     * (MVP 5.4 & 5.5 Logic)
     */
    @Transactional
    public void incrementNoShowCount(UUID userId) {
        // 1. Find user
        User user = userRepository.findById(userId)
                .orElseThrow(() -> new ResourceNotFoundException("User", "userId", userId));

        // 2. Increase counter
        int newNoShowCount = user.getNoShowCount() + 1;
        user.setNoShowCount(newNoShowCount);

        // 3. Block check (MVP 5.5)
        if (newNoShowCount >= NO_SHOW_LIMIT) {
            // Block user
            user.setStatus(UserStatus.suspended);
            user.setSuspendedUntil(OffsetDateTime.now().plusDays(3)); // 3-day block
            
            // Reset counter after applying the block
            user.setNoShowCount(0); 

            log.warn("User {} suspended for 3 days due to {} no-shows.", userId, newNoShowCount);
            
            // --- NEW LOGIC ---
            // 4. Award the "Unreliable" badge
            try {
                awardUnreliableBadge(user);
            } catch (Exception e) {
                log.error("Failed to award UNRELIABLE badge to user {}: {}", userId, e.getMessage());
                // Don't fail the transaction, just log the error
            }
            // --- END NEW LOGIC ---

        } else {
            log.info("User {} 'no-show' count incremented to {}.", userId, newNoShowCount);
        }

        // 5. Save user
        userRepository.save(user);
    }

    /**
     * Awards the "UNRELIABLE" badge to a user
     */
    private void awardUnreliableBadge(User user) {
        // Find the master "Unreliable" badge
        Badge unreliableBadge = badgeRepository.findByBadgeName(UNRELIABLE_BADGE_NAME)
                .orElseThrow(() -> new RuntimeException(UNRELIABLE_BADGE_NAME + " badge not found in database."));

        // Create the composite key
        UserBadgeId userBadgeId = new UserBadgeId();
        userBadgeId.setUserId(user.getUserId());
        userBadgeId.setBadgeId(unreliableBadge.getBadgeId());

        // Check if user already has this badge
        if (userBadgeRepository.existsById(userBadgeId)) {
            return; // Already has it, do nothing
        }

        // Assign the badge
        UserBadge newUserBadge = new UserBadge();
        newUserBadge.setId(userBadgeId);
        newUserBadge.setUser(user);
        newUserBadge.setBadge(unreliableBadge);
        
        userBadgeRepository.save(newUserBadge);
        log.info("Awarded {} badge to user {}", UNRELIABLE_BADGE_NAME, user.getUserId());
    }

    /**
     * Allows a user to block another user.
     * (MVP 4.2 Logic)
     */
    @Transactional
    public void blockUser(String blockerPhoneNumber, UUID blockedUserId) {
        // 1. Find blocker user
        User blockerUser = userRepository.findByPhoneNumber(blockerPhoneNumber)
                .orElseThrow(() -> new ResourceNotFoundException("User", "phoneNumber", blockerPhoneNumber));

        // 2. Find user to be blocked
        User blockedUser = userRepository.findById(blockedUserId)
                .orElseThrow(() -> new ResourceNotFoundException("User", "userId", blockedUserId));

        // 3. Prevent self-blocking
        if (blockerUser.getUserId().equals(blockedUser.getUserId())) {
            throw new IllegalArgumentException("A user cannot block themselves.");
        }

        // 4. Check if already blocked
        boolean alreadyBlocked = userBlockRepository.existsByBlockerUserIdAndBlockedUserId(
                blockerUser.getUserId(), blockedUser.getUserId());

        if (alreadyBlocked) {
            return;
        }

        // 5. Create UserBlock entity
        UserBlockId blockId = new UserBlockId();
        blockId.setBlockerUserId(blockerUser.getUserId());
        blockId.setBlockedUserId(blockedUser.getUserId());

        UserBlock newBlock = new UserBlock();
        newBlock.setId(blockId);
        newBlock.setBlocker(blockerUser);
        newBlock.setBlocked(blockedUser);

        // 6. Save block
        userBlockRepository.save(newBlock);
    }

    @Transactional
    public void updateFcmToken(String userPhoneNumber, UpdateFcmTokenDto dto) {
        User user = userRepository.findByPhoneNumber(userPhoneNumber)
                .orElseThrow(() -> new ResourceNotFoundException("User", "phoneNumber", userPhoneNumber));
        
        user.setPushNotificationToken(dto.fcmToken());
        userRepository.save(user);
        log.info("Updated FCM token for user: {}", user.getUserId());
    }

    @Transactional
    public void updateProfileVisibility(String userPhoneNumber, UpdateVisibilityDto dto) {
        User user = userRepository.findByPhoneNumber(userPhoneNumber)
                .orElseThrow(() -> new ResourceNotFoundException("User", "phoneNumber", userPhoneNumber));

        // Validation: You cannot set visibility to "same_gender" if you are not gender-verified.
        if (dto.visibility() == ProfileVisibility.same_gender && 
            user.getVerifiedGender() == VerifiedGender.not_verified) {
            
            throw new IllegalStateException("You must be gender-verified to use this privacy setting.");
        }

        user.setProfileVisibility(dto.visibility());
        userRepository.save(user);
        log.info("Updated profile visibility for user {} to {}", user.getUserId(), dto.visibility());
    }
}

package com.sahbifit.api.verification;

import java.util.Arrays;
import java.util.List;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.security.core.annotation.AuthenticationPrincipal;
import org.springframework.security.core.userdetails.UserDetails;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;
import org.springframework.web.multipart.MultipartFile;

import com.sahbifit.api.verification.dto.VerificationStatusDto;

import lombok.extern.slf4j.Slf4j;

@Slf4j
@RestController
@RequestMapping("/api/verification")
public class VerificationController {

    @Autowired
    private VerificationService verificationService;

    /**
     * Endpoint for submitting verification video.
     */
    @PostMapping("/submit")
    public ResponseEntity<?> submitVideo(
            @AuthenticationPrincipal UserDetails userDetails,
            @RequestParam("video") MultipartFile videoFile) {

        String userPhoneNumber = userDetails.getUsername();
        log.info("Video upload request from user: {}", userPhoneNumber);

        try {
            // 1. Validation: File is not empty
            if (videoFile.isEmpty()) {
                log.warn("Empty file from user: {}", userPhoneNumber);
                return ResponseEntity.badRequest().body("The video file is empty.");
            }

            // âœ… IMPROVED: Allow multiple video types for better UX
            String contentType = videoFile.getContentType();
            if (contentType == null || !contentType.startsWith("video/")) {
                log.warn("Invalid file type from user: {}, Type: {}", userPhoneNumber, contentType);
                return ResponseEntity.badRequest().body("Only video files are allowed.");
            }

            // âœ… OPTIONAL: More specific validation if needed
            List<String> allowedTypes = Arrays.asList("video/mp4", "video/quicktime", "video/webm", "video/x-msvideo");
            if (!allowedTypes.contains(contentType)) {
                log.warn("Unsupported video type from user: {}, Type: {}", userPhoneNumber, contentType);
                return ResponseEntity.badRequest().body("Only MP4, MOV, WebM and AVI video files are allowed.");
            }

            // âœ… REMOVED: File size validation (now handled by Spring configuration)
            // File size validation is now centralized via application.properties
            // spring.servlet.multipart.max-file-size=100MB
            // spring.servlet.multipart.max-request-size=100MB

            // 4. Processing
            verificationService.submitVerificationVideo(userPhoneNumber, videoFile);
            
            log.info("Video successfully processed for user: {}", userPhoneNumber);
            return ResponseEntity.ok("Video successfully submitted. Verification is pending.");

        } catch (IllegalStateException e) {
            log.warn("Invalid status for video upload from user: {}, Error: {}", userPhoneNumber, e.getMessage());
            return ResponseEntity.badRequest().body(e.getMessage());
        } catch (Exception e) {
            log.error("Error during video upload for user: {}", userPhoneNumber, e);
            return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR)
                    .body("Upload error: " + e.getMessage());
        }
    }

    /**
     * Endpoint for the app to query user status
     * (e.g., to display the "Probationary Mode" UI).
     */
    @GetMapping("/status")
    public ResponseEntity<VerificationStatusDto> getStatus(@AuthenticationPrincipal UserDetails userDetails) {
        String userPhoneNumber = userDetails.getUsername();
        
        try {
            VerificationStatusDto statusDto = verificationService.getVerificationStatus(userPhoneNumber);
            log.debug("Status query for user: {}, Status: {}", userPhoneNumber, statusDto.status());
            return ResponseEntity.ok(statusDto);
        } catch (Exception e) {
            log.error("Error during status query for user: {}", userPhoneNumber, e);
            return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).build();
        }
    }
}

package com.sahbifit.api.verification;

import java.io.IOException;
import java.time.OffsetDateTime;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.multipart.MultipartFile;

import com.sahbifit.api.external.storage.StorageService;
import com.sahbifit.api.user.model.User;
import com.sahbifit.api.user.repository.UserRepository;
import com.sahbifit.api.verification.dto.VerificationStatusDto;
import com.sahbifit.api.verification.model.VerificationStatus;
import com.sahbifit.api.verification.model.VerifiedGender;

import lombok.extern.slf4j.Slf4j;

@Slf4j
@Service
public class VerificationService {

    @Autowired
    private UserRepository userRepository;

    @Autowired
    private StorageService storageService; // CloudflareR2Service is injected

    /**
     * Processes the uploaded verification video of a user.
     */
    @Transactional
    public void submitVerificationVideo(String userPhoneNumber, MultipartFile videoFile) throws IOException {
        User user = userRepository.findByPhoneNumber(userPhoneNumber)
                .orElseThrow(() -> new RuntimeException("User not found."));

        // 1. Validation: Prevents duplicate uploads
        if (user.getVerificationStatus() == VerificationStatus.pending_review || 
            user.getVerificationStatus() == VerificationStatus.verified) {
            throw new IllegalStateException("Verification already completed or pending.");
        }

        // 1b. Get the key for the old video, if it exists
        String oldVideoKey = user.getVerificationVideoUrl();

        // 2. Generate unique filename
        String fileName = user.getUserId().toString() + "_" + System.currentTimeMillis() + ".mp4";
        String folder = "verification-videos";

        log.info("Starting video upload for user: {}, filename: {}", userPhoneNumber, fileName);

        // 3. Upload file - now returns the key instead of public URL
        String fileKey = storageService.uploadFile(videoFile, folder, fileName);

        // 4. Set user status to "pending_review" - store the key directly
        user.setVerificationVideoUrl(fileKey);
        user.setVerificationStatus(VerificationStatus.pending_review);

        // 5. Important: Reset gender if it was previously rejected
        user.setVerifiedGender(VerifiedGender.not_verified);

        // 6. Set the submission timestamp
        user.setVerificationSubmittedAt(OffsetDateTime.now());

        userRepository.save(user);

        log.info("Video successfully uploaded for user: {}, Key: {}", userPhoneNumber, fileKey);

        // 7. Delete the old file after the transaction is successful
        if (oldVideoKey != null && !oldVideoKey.equals(fileKey)) {
            try {
                storageService.deleteFile(oldVideoKey);
                log.info("Successfully deleted old verification video: {}", oldVideoKey);
            } catch (IOException e) {
                // Log this as a warning. The upload was still successful,
                // but we have an orphaned file in storage.
                log.warn("Failed to delete old video {}. New video {} was saved. This file is an orphan.", oldVideoKey, fileKey, e);
            }
        }
    }

    /**
     * Retrieves the current verification status of the user.
     */
    public VerificationStatusDto getVerificationStatus(String userPhoneNumber) {
        User user = userRepository.findByPhoneNumber(userPhoneNumber)
                .orElseThrow(() -> new RuntimeException("User not found."));

        return new VerificationStatusDto(user.getVerificationStatus(), user.getVerifiedGender());
    }
}
