    @Transactional
    public void register(RegisterRequest request) {
        // 1. reCAPTCHA v3 Validierung
        reCaptchaService.validate(request.getRecaptchaToken());

        // 2. Tunesische Nummernvalidierung
        if (!isTunisianNumber(request.getPhoneNumber())) {
            throw new IllegalArgumentException("UngÃ¼ltiges tunesisches Telefonnummernformat.");
        }

        // 3. PrÃ¼fen, ob der Benutzer bereits existiert
        if (userRepository.findByPhoneNumber(request.getPhoneNumber()).isPresent()) {
            throw new PhoneNumberAlreadyExistsException("Telefonnummer ist bereits vergeben.");
        }

        // 4. Benutzerobjekt erstellen
        User user = new User();
        user.setPhoneNumber(request.getPhoneNumber());
        user.setPassword(passwordEncoder.encode(request.getPassword()));
        user.setFirstName(request.getFirstName());
        user.setDateOfBirth(LocalDate.parse(request.getDateOfBirth()));

        // 5. Primary Gym setzen
        if (request.getPrimaryGymId() != null) {
            Gym primaryGym = gymRepository.findById(request.getPrimaryGymId())
                    .orElseThrow(() -> new ResourceNotFoundException("Gym nicht gefunden mit ID: " + request.getPrimaryGymId()));
            user.setPrimaryGym(primaryGym);
        }

        // 6. Profil-Tags (Goals, Schedule) verarbeiten und speichern
        java.util.Set<UserTag> allUserTags = new java.util.HashSet<>();

        // --- Verarbeitung der GOAL Tags (1 DB-Abfrage) ---
        if (request.getGoalTagIds() != null && !request.getGoalTagIds().isEmpty()) {
            
            // 1. Hole ALLE Goal-Tags mit EINER einzigen Datenbankabfrage
            java.util.List<Tag> goalTags = tagRepository.findAllById(request.getGoalTagIds());
            
            // 2. Validierung: Sicherstellen, dass alle angeforderten IDs auch gefunden wurden
            if (goalTags.size() != request.getGoalTagIds().size()) {
                throw new ResourceNotFoundException("Ein oder mehrere 'Goal' Tags wurden nicht gefunden.");
            }

            // 3. Mappe die Tags im Speicher (keine weiteren DB-Aufrufe)
            java.util.Set<UserTag> goalUserTags = goalTags.stream()
                .map(tag -> {
                    // Optionale, aber empfohlene Validierung des Tag-Typs
                    if (tag.getTagType() != com.sahbifit.api.model.enums.TagType.goal) {
                         throw new IllegalArgumentException("Tag " + tag.getTagName() + " ist kein 'Goal' Tag.");
                    }
                    UserTag userTag = new UserTag();
                    userTag.setUser(user); // VerknÃ¼pft mit dem neuen User
                    userTag.setTag(tag);   // VerknÃ¼pft mit dem gefundenen Tag
                    return userTag;
                })
                .collect(Collectors.toSet());
            
            allUserTags.addAll(goalUserTags);
        }

        // --- Verarbeitung der SCHEDULE Tags (1 DB-Abfrage) ---
        if (request.getScheduleTagIds() != null && !request.getScheduleTagIds().isEmpty()) {

            // 1. Hole ALLE Schedule-Tags mit EINER einzigen Datenbankabfrage
            java.util.List<Tag> scheduleTags = tagRepository.findAllById(request.getScheduleTagIds());
            
            // 2. Validierung
            if (scheduleTags.size() != request.getScheduleTagIds().size()) {
                throw new ResourceNotFoundException("Ein oder mehrere 'Schedule' Tags wurden nicht gefunden.");
            }

            // 3. Mappe die Tags im Speicher (keine weiteren DB-Aufrufe)
            java.util.Set<UserTag> scheduleUserTags = scheduleTags.stream()
                .map(tag -> {
                    // Validierung des Tag-Typs
                    if (tag.getTagType() != com.sahbifit.api.model.enums.TagType.schedule) {
                         throw new IllegalArgumentException("Tag " + tag.getTagName() + " ist kein 'Schedule' Tag.");
                    }
                    UserTag userTag = new UserTag();
                    userTag.setUser(user);
                    userTag.setTag(tag);
                    return userTag;
                })
                .collect(Collectors.toSet());

            allUserTags.addAll(scheduleUserTags);
        }

        // Hier rufen wir die EINE korrekte Set-Methode von User.java auf
        user.setUserTags(allUserTags);

        // 7. Standardwerte setzen (gemÃ¤ÃŸ Ihrer Spezifikation)
        user.setRole(UserRole.member);
        user.setStatus(UserStatus.active);
        user.setVerificationStatus(VerificationStatus.not_started);
        user.setVerifiedGender(VerifiedGender.not_verified);
        user.setPhoneVerified(false);

        // 8. OTP (One-Time Password) fÃ¼r SMS generieren
        String otp = generateOtp();
        
        // 9. OTP und Ablaufdatum in der DB speichern
        user.setSmsOtp(otp);
        user.setSmsOtpExpiresAt(OffsetDateTime.now().plusMinutes(10)); // 10 Minuten GÃ¼ltigkeit
        
        // 10. Benutzer speichern
        userRepository.save(user);

        // 11. SMS senden
        smsService.sendVerificationSms(user.getPhoneNumber(), otp);
    }

    public AuthResponse login(LoginRequest loginRequest) {
        // REMOVED: The manual user check and phone verification check
        // All validation will now happen in UserDetailsServiceImpl
        
        // 1. Authenticate (this will call UserDetailsServiceImpl.loadUserByUsername)
        Authentication authentication = authenticationManager.authenticate(
                new UsernamePasswordAuthenticationToken(
                        loginRequest.getPhoneNumber(),
                        loginRequest.getPassword()
                )
        );

        // 2. Security Context setzen
        SecurityContextHolder.getContext().setAuthentication(authentication);

        // 3. JWT generieren
        String jwt = tokenProvider.generateToken(authentication);
        return new AuthResponse(jwt);
    }

    /**
     * Verifiziert den SMS-OTP-Code eines Benutzers.
     */
    @Transactional
    public void verifySmsOtp(VerifySmsRequest request) {
        // 1. Benutzer anhand der Telefonnummer finden
        User user = userRepository.findByPhoneNumber(request.getPhoneNumber())
                .orElseThrow(() -> new ResourceNotFoundException("Benutzer nicht gefunden."));

        // 2. PrÃ¼fen, ob bereits verifiziert
        if (user.isPhoneVerified()) {
            // Kein Fehler, aber auch keine Aktion nÃ¶tig.
            return; 
        }

        // 3. PrÃ¼fen, ob ein OTP gespeichert ist
        if (user.getSmsOtp() == null || user.getSmsOtpExpiresAt() == null) {
            throw new IllegalStateException("Kein ausstehender OTP gefunden. Bitte erneut anfordern.");
        }

        // 4. PrÃ¼fen, ob der OTP abgelaufen ist
        if (OffsetDateTime.now().isAfter(user.getSmsOtpExpiresAt())) {
            // OTP lÃ¶schen, um Wiederverwendung zu verhindern
            user.setSmsOtp(null);
            user.setSmsOtpExpiresAt(null);
            userRepository.save(user);
            throw new OtpExpiredException("OTP ist abgelaufen. Bitte einen neuen Code anfordern.");
        }

        // 5. PrÃ¼fen, ob der OTP-Code korrekt ist
        if (!user.getSmsOtp().equals(request.getOtp())) {
            throw new InvalidOtpException("UngÃ¼ltiger OTP-Code.");
        }

        // 6. Erfolg! Benutzer als verifiziert markieren
        user.setPhoneVerified(true);
        
        // 7. OTP-Felder aus SicherheitsgrÃ¼nden lÃ¶schen
        user.setSmsOtp(null);
        user.setSmsOtpExpiresAt(null);

        userRepository.save(user);
    }

    /**
     * Sendet einen neuen OTP-Code an eine nicht verifizierte Telefonnummer.
     * Wird verwendet, wenn der ursprÃ¼ngliche OTP abgelaufen ist.
     */
    @Transactional
    public void resendOtp(ResendOtpRequest request) {
        // 1. reCAPTCHA validieren
        reCaptchaService.validate(request.getRecaptchaToken());

        // 2. Tunesische Nummernvalidierung
        if (!isTunisianNumber(request.getPhoneNumber())) {
            throw new IllegalArgumentException("UngÃ¼ltiges tunesisches Telefonnummernformat.");
        }

        // 3. Benutzer anhand der Telefonnummer finden
        User user = userRepository.findByPhoneNumber(request.getPhoneNumber())
                .orElseThrow(() -> new ResourceNotFoundException("Benutzer nicht gefunden."));

        // 4. PrÃ¼fen, ob bereits verifiziert
        if (user.isPhoneVerified()) {
            throw new IllegalArgumentException("Konto ist bereits verifiziert. Bitte einloggen.");
        }

        // 5. Neuen OTP generieren
        String newOtp = generateOtp();
        
        // 6. Neuen OTP und Ablaufdatum setzen
        user.setSmsOtp(newOtp);
        user.setSmsOtpExpiresAt(OffsetDateTime.now().plusMinutes(10)); // 10 Minuten GÃ¼ltigkeit
        
        // 7. Benutzer speichern
        userRepository.save(user);

        // 8. SMS mit neuem OTP senden
        smsService.sendVerificationSms(user.getPhoneNumber(), newOtp);
    }

    // --- Private Hilfsmethoden ---

    private boolean isTunisianNumber(String phone) {
        // Implementieren Sie hier Ihre Validierungslogik
        // z.B. Regex fÃ¼r (2, 3, 4, 5, 7, 9) + 7 Ziffern
        return phone != null && phone.matches("^(2|3|4|5|7|9)\\d{7}$");
    }

    private String generateOtp() {
        // Erzeugt einen 6-stelligen Code
        return String.format("%06d", new Random().nextInt(999999));
    }
}

package com.sahbifit.api.external.recaptcha;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import org.springframework.util.LinkedMultiValueMap;
import org.springframework.util.MultiValueMap;
import org.springframework.web.client.RestTemplate;

import com.sahbifit.api.config.ReCaptchaConfig;

@Service
public class GoogleReCaptchaService implements ReCaptchaService {

    private static final Logger logger = LoggerFactory.getLogger(GoogleReCaptchaService.class);

    private final ReCaptchaConfig reCaptchaConfig;
    private final RestTemplate restTemplate;

    @Autowired
    public GoogleReCaptchaService(ReCaptchaConfig reCaptchaConfig, RestTemplate restTemplate) {
        this.reCaptchaConfig = reCaptchaConfig;
        // Wir verwenden das RestTemplate-Bean wieder, das in WinSmsConfig erstellt wurde!
        this.restTemplate = restTemplate; 
    }

    @Override
    public void validate(String token) {
        // Google erwartet eine POST-Anfrage mit 'x-www-form-urlencoded' Daten
        MultiValueMap<String, String> body = new LinkedMultiValueMap<>();
        body.add("secret", reCaptchaConfig.getSecretKey());
        body.add("response", token);
        // Optional kÃ¶nnen Sie auch die IP des Benutzers mitsenden:
        // body.add("remoteip", userIp);

        ReCaptchaResponse response;
        try {
            response = restTemplate.postForObject(
                    reCaptchaConfig.getVerifyUrl(),
                    body,
                    ReCaptchaResponse.class
            );
        } catch (Exception e) {
            logger.error("Fehler bei der Kommunikation mit der Google reCAPTCHA API: {}", e.getMessage());
            // Wir werfen eine Exception, um die Registrierung zu stoppen
            throw new RuntimeException("reCAPTCHA-Dienst nicht erreichbar.");
        }

        // 1. PrÃ¼fen, ob die Anfrage selbst erfolgreich war
        if (response == null || !response.isSuccess()) {
            logger.warn("reCAPTCHA-Validierung fehlgeschlagen. Fehler: {}", 
                (response != null ? response.getErrorCodes() : "Keine Antwort"));
            throw new RuntimeException("reCAPTCHA-Validierung fehlgeschlagen.");
        }

        // 2. PrÃ¼fen, ob der Score (Bot-Wahrscheinlichkeit) ausreichend ist
        if (response.getScore() < reCaptchaConfig.getScoreThreshold()) {
            logger.warn("Potenzieller Bot blockiert. reCAPTCHA Score: {}", response.getScore());
            throw new RuntimeException("Validierung fehlgeschlagen (niedriger Score).");
        }
        
        // 3. (Optional) Sie kÃ¶nnen auch die 'action' prÃ¼fen, 
        //    um sicherzustellen, dass das Token von Ihrer Registrierungsseite stammt.
        //    z.B. if (!"register".equals(response.getAction())) { ... }

        logger.info("reCAPTCHA-Validierung erfolgreich (Score: {}).", response.getScore());
    }
}

package com.sahbifit.api.external.storage;

import lombok.extern.slf4j.Slf4j;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import org.springframework.web.multipart.MultipartFile;

import software.amazon.awssdk.core.sync.RequestBody;
import software.amazon.awssdk.services.s3.S3Client;
import software.amazon.awssdk.services.s3.model.DeleteObjectRequest;
import software.amazon.awssdk.services.s3.model.PutObjectRequest;
import software.amazon.awssdk.services.s3.model.S3Exception;

import java.io.IOException;
import java.util.UUID;

import com.sahbifit.api.config.CloudflareConfig;

@Slf4j
@Service
public class CloudflareR2Service implements StorageService {

    @Autowired
    private S3Client s3Client;
    
    @Autowired
    private CloudflareConfig cloudflareConfig;

    @Override
    public String uploadFile(MultipartFile file, String folder, String fileName) throws IOException {
        try {
            // Validierung der Datei
            if (file.isEmpty()) {
                throw new IllegalArgumentException("Datei ist leer");
            }

            // Generiere den vollstÃ¤ndigen S3 Key (Pfad im Bucket)
            String key = generateFileKey(folder, fileName);
            
            log.info("Lade Datei hoch: {} -> {}/{}", file.getOriginalFilename(), 
                    cloudflareConfig.getBucketName(), key);

            // Metadaten fÃ¼r das Video
            var metadata = new java.util.HashMap<String, String>();
            metadata.put("Content-Type", file.getContentType());
            metadata.put("Original-Filename", file.getOriginalFilename());
            metadata.put("Upload-Timestamp", String.valueOf(System.currentTimeMillis()));

            // Erstelle PutObjectRequest
            PutObjectRequest putObjectRequest = PutObjectRequest.builder()
                    .bucket(cloudflareConfig.getBucketName())
                    .key(key)
                    .contentType(file.getContentType())
                    .contentLength(file.getSize())
                    .metadata(metadata)
                    .build();

            // Datei hochladen
            s3Client.putObject(putObjectRequest, 
                    RequestBody.fromBytes(file.getBytes()));

            // Generiere die Ã¶ffentliche URL
            String publicUrl = generatePublicUrl(key);
            
            log.info("Datei erfolgreich hochgeladen: {}", publicUrl);
            return publicUrl;

        } catch (S3Exception e) {
            log.error("S3 Fehler beim Hochladen der Datei: {}", e.awsErrorDetails().errorMessage(), e);
            throw new IOException("Fehler beim Hochladen der Datei: " + e.awsErrorDetails().errorMessage(), e);
        } catch (Exception e) {
            log.error("Unerwarteter Fehler beim Hochladen der Datei", e);
            throw new IOException("Unerwarteter Fehler beim Hochladen: " + e.getMessage(), e);
        }
    }

    @Override
    public void deleteFile(String fileUrl) throws IOException {
        try {
            // Extrahiere den Key aus der URL
            String key = extractKeyFromUrl(fileUrl);
            
            log.info("LÃ¶sche Datei: {}", key);
            
            DeleteObjectRequest deleteObjectRequest = DeleteObjectRequest.builder()
                    .bucket(cloudflareConfig.getBucketName())
                    .key(key)
                    .build();

            s3Client.deleteObject(deleteObjectRequest);
            
            log.info("Datei erfolgreich gelÃ¶scht: {}", key);

        } catch (S3Exception e) {
            log.error("S3 Fehler beim LÃ¶schen der Datei: {}", e.awsErrorDetails().errorMessage(), e);
            throw new IOException("Fehler beim LÃ¶schen der Datei: " + e.awsErrorDetails().errorMessage(), e);
        }
    }

    /**
     * Generiert den S3 Key fÃ¼r die Datei
     */
    private String generateFileKey(String folder, String fileName) {
        // Entferne fÃ¼hrende und nachfolgende SchrÃ¤gstriche
        String cleanFolder = folder.replaceAll("^/|/$", "");
        return String.format("%s/%s", cleanFolder, fileName);
    }

    /**
     * Generiert die Ã¶ffentliche URL fÃ¼r die Datei
     */
    private String generatePublicUrl(String key) {
        return String.format("%s/%s", cloudflareConfig.getPublicUrl(), key);
    }

    /**
     * Extrahiert den S3 Key aus der Ã¶ffentlichen URL
     */
    private String extractKeyFromUrl(String fileUrl) {
        String baseUrl = cloudflareConfig.getPublicUrl() + "/";
        if (fileUrl.startsWith(baseUrl)) {
            return fileUrl.substring(baseUrl.length());
        }
        throw new IllegalArgumentException("UngÃ¼ltige R2 URL: " + fileUrl);
    }
    
    /**
     * Hilfsmethode zum Generieren eines eindeutigen Dateinamens
     */
    public String generateUniqueFileName(String originalFileName) {
        String extension = "";
        if (originalFileName != null && originalFileName.contains(".")) {
            extension = originalFileName.substring(originalFileName.lastIndexOf("."));
        }
        return UUID.randomUUID().toString() + extension;
    }
}

            String jwt = getJwtFromRequest(request);

            if (StringUtils.hasText(jwt) && tokenProvider.validateToken(jwt)) {
                String username = tokenProvider.getUsernameFromJWT(jwt); // Dies ist die Telefonnummer

                // ðŸ†• OPTION 1: Rollen direkt aus Token lesen (schneller, keine DB-Abfrage)
                List<String> roles = tokenProvider.getRolesFromJWT(jwt);
                List<GrantedAuthority> authorities = roles.stream()
                        .map(SimpleGrantedAuthority::new)
                        .collect(Collectors.toList());
                
                UsernamePasswordAuthenticationToken authentication = 
                    new UsernamePasswordAuthenticationToken(username, null, authorities);
                authentication.setDetails(new WebAuthenticationDetailsSource().buildDetails(request));

                SecurityContextHolder.getContext().setAuthentication(authentication);
                
                // ðŸ†• OPTION 2: Original-Logik (mit DB-Abfrage fÃ¼r aktuelle User-Daten)
                // UserDetails userDetails = userDetailsServiceImpl.loadUserByUsername(username);
                // UsernamePasswordAuthenticationToken authentication = 
                //     new UsernamePasswordAuthenticationToken(userDetails, null, userDetails.getAuthorities());
                // authentication.setDetails(new WebAuthenticationDetailsSource().buildDetails(request));
                // SecurityContextHolder.getContext().setAuthentication(authentication);
            }
        } catch (Exception ex) {
            logger.error("Could not set user authentication in security context", ex);
        }

        filterChain.doFilter(request, response);
    }

    private String getJwtFromRequest(HttpServletRequest request) {
        String bearerToken = request.getHeader("Authorization");
        if (StringUtils.hasText(bearerToken) && bearerToken.startsWith("Bearer ")) {
            return bearerToken.substring(7, bearerToken.length());
        }
        return null;

package com.sahbifit.api.security;

import java.util.Date;
import java.util.List;
import java.util.stream.Collectors;

import javax.crypto.SecretKey;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.security.core.Authentication;
import org.springframework.security.core.GrantedAuthority;
import org.springframework.security.core.userdetails.UserDetails;
import org.springframework.stereotype.Component;

import io.jsonwebtoken.Claims;
import io.jsonwebtoken.ExpiredJwtException;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.MalformedJwtException;
import io.jsonwebtoken.SignatureAlgorithm;
import io.jsonwebtoken.SignatureException;
import io.jsonwebtoken.UnsupportedJwtException;
import io.jsonwebtoken.security.Keys;

@Component
public class JwtTokenProvider {

    private static final Logger logger = LoggerFactory.getLogger(JwtTokenProvider.class);

    @Value("${app.jwtSecret}")
    private String jwtSecret;

    @Value("${app.jwtExpirationInMs}")
    private int jwtExpirationInMs;

    private SecretKey getSigningKey() {
        return Keys.hmacShaKeyFor(jwtSecret.getBytes());
    }

    // Erstellt ein JWT fÃ¼r einen authentifizierten Benutzer MIT ROLEN
    public String generateToken(Authentication authentication) {
        UserDetails userPrincipal = (UserDetails) authentication.getPrincipal();

        // Rollen aus den Authorities extrahieren
        List<String> roles = userPrincipal.getAuthorities().stream()
                .map(GrantedAuthority::getAuthority)
                .collect(Collectors.toList());

        Date now = new Date();
        Date expiryDate = new Date(now.getTime() + jwtExpirationInMs);

        return Jwts.builder()
                .setSubject(userPrincipal.getUsername()) // Telefonnummer
                .claim("roles", roles) // ðŸ†• Rollen im Token speichern
                .setIssuedAt(new Date())
                .setExpiration(expiryDate)
                .signWith(getSigningKey(), SignatureAlgorithm.HS512)
                .compact();
    }

    // Extrahiert die Telefonnummer aus dem JWT
    public String getUsernameFromJWT(String token) {
        Claims claims = Jwts.parserBuilder()
                .setSigningKey(getSigningKey())
                .build()
                .parseClaimsJws(token)
                .getBody();

        return claims.getSubject();
    }

    // ðŸ†• Extrahiert Rollen aus dem JWT Token
    public List<String> getRolesFromJWT(String token) {
        Claims claims = Jwts.parserBuilder()
                .setSigningKey(getSigningKey())
                .build()
                .parseClaimsJws(token)
                .getBody();

        return claims.get("roles", List.class);
    }

    // Validiert ein JWT
    public boolean validateToken(String authToken) {
        try {
            Jwts.parserBuilder().setSigningKey(getSigningKey()).build().parseClaimsJws(authToken);
            return true;
        } catch (SignatureException ex) {
            logger.error("Invalid JWT signature");
        } catch (MalformedJwtException ex) {
            logger.error("Invalid JWT token");
        } catch (ExpiredJwtException ex) {
            logger.error("Expired JWT token");
        } catch (UnsupportedJwtException ex) {
            logger.error("Unsupported JWT token");
        } catch (IllegalArgumentException ex) {
            logger.error("JWT claims string is empty.");
        }
        return false;
    }
}

package com.sahbifit.api.security;

import java.util.Arrays;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.authentication.AuthenticationManager;
import org.springframework.security.config.annotation.authentication.configuration.AuthenticationConfiguration;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.web.SecurityFilterChain;
import org.springframework.security.web.authentication.UsernamePasswordAuthenticationFilter;
import org.springframework.web.cors.CorsConfiguration;
import org.springframework.web.cors.CorsConfigurationSource;
import org.springframework.web.cors.UrlBasedCorsConfigurationSource;

@Configuration
@EnableWebSecurity
public class SecurityConfig {

    @Autowired
    private JwtAuthenticationFilter jwtAuthenticationFilter;

    @Bean
    public SecurityFilterChain filterChain(HttpSecurity http) throws Exception {
        http
            .cors(cors -> cors.configurationSource(corsConfigurationSource()))
            .csrf(csrf -> csrf.disable())
            .sessionManagement(session -> session.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
            .authorizeHttpRequests(authorize -> authorize
                // Ã–ffentliche Endpunkte
                .requestMatchers("/api/auth/**").permitAll()
                
                // --- ADMIN ENDPUNKTE ---
                .requestMatchers("/api/admin/**").hasRole("ADMIN")
                
                // --- VERIFICATION ENDPUNKTE - fÃ¼r alle authentifizierten Benutzer ---
                .requestMatchers("/api/verification/**").authenticated()
                
                // Alle anderen Anfragen erfordern Authentifizierung
                .anyRequest().authenticated()
            );

        http.addFilterBefore(jwtAuthenticationFilter, UsernamePasswordAuthenticationFilter.class);
        return http.build();
    }

    @Bean
    public CorsConfigurationSource corsConfigurationSource() {
        CorsConfiguration configuration = new CorsConfiguration();
        configuration.setAllowedOriginPatterns(Arrays.asList("*"));
        configuration.setAllowedMethods(Arrays.asList("GET", "POST", "PUT", "DELETE", "OPTIONS"));
        configuration.setAllowedHeaders(Arrays.asList("*"));
        configuration.setAllowCredentials(true);
        
        UrlBasedCorsConfigurationSource source = new UrlBasedCorsConfigurationSource();
        source.registerCorsConfiguration("/**", configuration);
        return source;
    }

    @Bean
    public AuthenticationManager authenticationManager(AuthenticationConfiguration config) throws Exception {
        return config.getAuthenticationManager();
    }

    @Bean
    public PasswordEncoder passwordEncoder() {
        return new BCryptPasswordEncoder();
    }
}

package com.sahbifit.api.security;

import java.util.ArrayList;
import java.util.List;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.security.core.GrantedAuthority;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.core.userdetails.UserDetails;
import org.springframework.security.core.userdetails.UserDetailsService;
import org.springframework.security.core.userdetails.UsernameNotFoundException;
import org.springframework.stereotype.Service;

import com.sahbifit.api.user.model.User;
import com.sahbifit.api.user.repository.UserRepository;

@Service
public class UserDetailsServiceImpl implements UserDetailsService {

    @Autowired
    private UserRepository userRepository;

    @Override
    public UserDetails loadUserByUsername(String phoneNumber) throws UsernameNotFoundException {
        User user = userRepository.findByPhoneNumber(phoneNumber)
                .orElseThrow(() -> new UsernameNotFoundException("User not found with phone number: " + phoneNumber));

        // âœ… Moved phone verification check here from AuthService
        if (!user.isPhoneVerified()) {
            throw new UsernameNotFoundException("User phone not verified."); 
        }

        // Convert UserRole to Spring Security Role
        List<GrantedAuthority> authorities = new ArrayList<>();
        String roleName = "ROLE_" + user.getRole().name().toUpperCase();
        authorities.add(new SimpleGrantedAuthority(roleName));

        return new org.springframework.security.core.userdetails.User(
                user.getPhoneNumber(),
                user.getPassword(),
                authorities
        );
    }
}

package com.sahbifit.api.verification;

import java.util.Arrays;
import java.util.List;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.security.core.annotation.AuthenticationPrincipal;
import org.springframework.security.core.userdetails.UserDetails;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;
import org.springframework.web.multipart.MultipartFile;

import com.sahbifit.api.verification.dto.VerificationStatusDto;

import lombok.extern.slf4j.Slf4j;

@Slf4j
@RestController
@RequestMapping("/api/verification")
public class VerificationController {

    @Autowired
    private VerificationService verificationService;

    /**
     * Endpunkt zum Einreichen des Verifizierungsvideos.
     */
    @PostMapping("/submit")
    public ResponseEntity<?> submitVideo(
            @AuthenticationPrincipal UserDetails userDetails,
            @RequestParam("video") MultipartFile videoFile) {

        String userPhoneNumber = userDetails.getUsername();
        log.info("Video-Upload-Anfrage von Benutzer: {}", userPhoneNumber);

        try {
            // 1. Validierung: Datei ist nicht leer
            if (videoFile.isEmpty()) {
                log.warn("Leere Datei von Benutzer: {}", userPhoneNumber);
                return ResponseEntity.badRequest().body("Die Videodatei ist leer.");
            }

            // âœ… IMPROVED: Allow multiple video types for better UX
            String contentType = videoFile.getContentType();
            if (contentType == null || !contentType.startsWith("video/")) {
                log.warn("UngÃ¼ltiger Dateityp von Benutzer: {}, Typ: {}", userPhoneNumber, contentType);
                return ResponseEntity.badRequest().body("Nur Videodateien sind erlaubt.");
            }

            // âœ… OPTIONAL: More specific validation if needed
            List<String> allowedTypes = Arrays.asList("video/mp4", "video/quicktime", "video/webm", "video/x-msvideo");
            if (!allowedTypes.contains(contentType)) {
                log.warn("Nicht unterstÃ¼tzter Videotyp von Benutzer: {}, Typ: {}", userPhoneNumber, contentType);
                return ResponseEntity.badRequest().body("Nur MP4, MOV, WebM und AVI Videodateien sind erlaubt.");
            }

            // 3. Validierung: DateigrÃ¶ÃŸe (max. 100MB)
            if (videoFile.getSize() > 100 * 1024 * 1024) {
                log.warn("Datei zu groÃŸ von Benutzer: {}, GrÃ¶ÃŸe: {} bytes", userPhoneNumber, videoFile.getSize());
                return ResponseEntity.badRequest().body("Die Videodatei darf maximal 100MB groÃŸ sein.");
            }

            // 4. Verarbeitung
            verificationService.submitVerificationVideo(userPhoneNumber, videoFile);
            
            log.info("Video erfolgreich verarbeitet fÃ¼r Benutzer: {}", userPhoneNumber);
            return ResponseEntity.ok("Video erfolgreich eingereicht. Die ÃœberprÃ¼fung steht aus.");

        } catch (IllegalStateException e) {
            log.warn("UngÃ¼ltiger Status fÃ¼r Video-Upload von Benutzer: {}, Fehler: {}", userPhoneNumber, e.getMessage());
            return ResponseEntity.badRequest().body(e.getMessage());
        } catch (Exception e) {
            log.error("Fehler beim Video-Upload fÃ¼r Benutzer: {}", userPhoneNumber, e);
            return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR)
                    .body("Fehler beim Upload: " + e.getMessage());
        }
    }

    /**
     * Endpunkt fÃ¼r die App, um den Status des Benutzers abzufragen
     * (z.B. um den "Probationary Mode" UI anzuzeigen).
     */
    @GetMapping("/status")
    public ResponseEntity<VerificationStatusDto> getStatus(@AuthenticationPrincipal UserDetails userDetails) {
        String userPhoneNumber = userDetails.getUsername();
        
        try {
            VerificationStatusDto statusDto = verificationService.getVerificationStatus(userPhoneNumber);
            log.debug("Statusabfrage fÃ¼r Benutzer: {}, Status: {}", userPhoneNumber, statusDto.getStatus());
            return ResponseEntity.ok(statusDto);
        } catch (Exception e) {
            log.error("Fehler bei Statusabfrage fÃ¼r Benutzer: {}", userPhoneNumber, e);
            return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).build();
        }
    }
}

package com.sahbifit.api.verification;

import java.io.IOException;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.multipart.MultipartFile;

import com.sahbifit.api.external.storage.StorageService;
import com.sahbifit.api.model.enums.VerificationStatus;
import com.sahbifit.api.model.enums.VerifiedGender;
import com.sahbifit.api.user.model.User;
import com.sahbifit.api.user.repository.UserRepository;
import com.sahbifit.api.verification.dto.VerificationStatusDto;

import lombok.extern.slf4j.Slf4j;

@Slf4j
@Service
public class VerificationService {

    @Autowired
    private UserRepository userRepository;

    @Autowired
    private StorageService storageService; // CloudflareR2Service wird injiziert

    /**
     * Verarbeitet das hochgeladene Verifizierungsvideo eines Benutzers.
     */
    @Transactional
    public void submitVerificationVideo(String userPhoneNumber, MultipartFile videoFile) throws IOException {
        User user = userRepository.findByPhoneNumber(userPhoneNumber)
                .orElseThrow(() -> new RuntimeException("Benutzer nicht gefunden."));

        // 1. Validierung: Verhindert doppelte Uploads
        if (user.getVerificationStatus() == VerificationStatus.pending_review || 
            user.getVerificationStatus() == VerificationStatus.verified) {
            throw new IllegalStateException("ÃœberprÃ¼fung bereits abgeschlossen oder ausstehend.");
        }

        // 2. Eindeutigen Dateinamen generieren
        String fileName = user.getUserId().toString() + "_" + System.currentTimeMillis() + ".mp4";
        String folder = "verification-videos";

        log.info("Starte Video-Upload fÃ¼r Benutzer: {}, Dateiname: {}", userPhoneNumber, fileName);

        // 3. Datei hochladen
        String fileUrl = storageService.uploadFile(videoFile, folder, fileName);

        // 4. Benutzerstatus auf "pending_review" setzen
        user.setVerificationVideoUrl(fileUrl);
        user.setVerificationStatus(VerificationStatus.pending_review);

        // 5. Wichtig: Geschlecht zurÃ¼cksetzen, falls es zuvor abgelehnt wurde
        user.setVerifiedGender(VerifiedGender.not_verified);

        userRepository.save(user);
        
        log.info("Video erfolgreich hochgeladen fÃ¼r Benutzer: {}, URL: {}", userPhoneNumber, fileUrl);
    }

    /**
     * Ruft den aktuellen Verifizierungsstatus des Benutzers ab.
     */
    public VerificationStatusDto getVerificationStatus(String userPhoneNumber) {
        User user = userRepository.findByPhoneNumber(userPhoneNumber)
                .orElseThrow(() -> new RuntimeException("Benutzer nicht gefunden."));

        return new VerificationStatusDto(user.getVerificationStatus(), user.getVerifiedGender());
    }
}
